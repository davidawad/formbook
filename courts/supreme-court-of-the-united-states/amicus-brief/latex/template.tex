% Supreme Court of the United States: amicus curiae brief (booklet format), LaTeX engine.
% Same rule parameters as typst/scotus-amicus-brief-amicus-brief.typ (Rules 33.1, 34, 37).
% The body and every front-matter block are produced by latex/brief.lua as raw LaTeX; this file
% holds the page, type and macro definitions.
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, contents rows) is the shared package lib/latex/xengine.sty, which fb render
% inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are lib/latex/xengine.lua, run
% by brief.lua. Every size and space here is in bp (12 bp = typst's 12 pt), and blocks are spaced with \addvspace
% (the larger of the space below one and above the next).
\documentclass[11pt]{article}
\usepackage[T1]{fontenc}
% Rule 33.1(b): Century family. docs/FONTS.md substitute: TeX Gyre Schola (same face as typst/). tgschola ships with
% every TeX Live (package tex-gyre), so the generated .tex compiles with stock pdflatex and no repo-local file.
\usepackage{tgschola}
% Rule 33.1(c): the text field 4 1/8 by 7 1/8 in. No running head: its height and separation are zero here, in
% geometry, so the top margin is the text's. The page number sits where typst sets it.
\usepackage[paperwidth=6.125in,paperheight=9.25in,left=1in,right=1in,top=1in,bottom=1.125in,footskip=36.3bp,headheight=0pt,headsep=0pt]{geometry}
\usepackage[hidelinks,breaklinks]{hyperref}
\usepackage{graphicx}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
\usepackage{xeschola} % lib/latex/xeschola.sty: Schola at the advances of the vendored font, inlined by fb render
\makeatletter

% Rule 33.1(b): 12-point type with 2 points of leading: 14 bp line boxes; notes 10 bp in 12 bp boxes.
\renewcommand\normalsize{\xe@lines{12}{14}}
\renewcommand\footnotesize{\xe@lines{10}{12}}
\normalsize
\setlength{\parindent}{0.4in} % after xengine.sty's \raggedright, which sets it to zero
\pagestyle{plain}
\setcounter{secnumdepth}{0}

% Footnotes [footnote.entry]: 12 bp clear of the text, a 0.9 in rule, 6 bp between notes; the mark (58 % of the
% 10 bp note type, raised 29 %) and a quarter em before the text. LaTeX resets the right skip inside a footnote:
% it is set back here.
\xe@footnotemarks{12}
\setlength{\skip\footins}{12bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.9in height 0.5bp\kern-0.25bp}
\newcommand\scotus@notemark[1]{\raisebox{2.9bp}{\fontsize{5.8bp}{0bp}\selectfont #1}}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=6bp\relax
  \vskip 0pt\noindent\scotus@notemark{\@thefnmark}\hskip 0.25em #1}

% Block quotations over 50 words are indented on both sides (Rule 33.1(b)), 12 bp above and below.
\renewenvironment{quote}{%
  \par\addvspace{12bp}%
  \begingroup\leftskip=0.4in\rightskip=\dimexpr0.4in\relax plus 1fil\relax\parindent=0pt\relax
}{%
  \par\endgroup\addvspace{12bp}%
}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% The page number: centered, in the 14 bp line box.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\normalsize\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}

% Space between two paragraphs of the body (one em; a heading's own space wins next to one, as in typst).
\newcommand\scotuspar{\par\addvspace{12bp}}

% Headings (typst: style-heading). A heading never ends a page; the paragraph after it is not indented.
\@clubpenalty=\clubpenalty % \@afterheading restores the club penalty from it: no orphans (xengine.sty)
% Section: centred bold capitals, 16.8 bp above, 12 bp below.
\newcommand{\scotussection}[2]{%
  \par\addvspace{16.8bp}%
  \begingroup\centering\bfseries\leavevmode#2\MakeUppercase{#1}\par\endgroup
  \nobreak\addvspace{12bp}\@afterindentfalse\@afterheading}
% Points: #1 indent, #2 printed label (in a 0.4 in box), #3 text hanging under the text, #4 label; 12 / 10.8 bp.
\newcommand{\scotuspoint}[4]{%
  \par\addvspace{12bp}%
  \begingroup\noindent\hangindent=\dimexpr#1+0.4in\relax\hangafter=1\relax#4%
  \hspace*{#1}\makebox[0.4in][l]{\textbf{#2}}\textbf{#3}\par\endgroup
  \nobreak\addvspace{10.8bp}\@afterindentfalse\@afterheading}

% Table of contents rows (Rule 34.2; typst: toc-row, listing-row): 6 bp above, the text wraps before a 24 bp page
% column, a dot leader runs to the page. #1 indent, #2 label (empty for a section), #3 title, #4 page.
\newcommand{\tocsection}[2]{%
  \begingroup\setlength{\xe@col}{24bp}\xe@row{6bp}{0pt}{\MakeUppercase{#1}}{#2}\endgroup}
\newcommand{\tocpoint}[4]{%
  \begingroup\setlength{\xe@col}{24bp}%
  \xe@row{6bp}{\dimexpr#1+0.4in\relax}{\hspace*{#1}\makebox[0.4in][l]{#2}#3}{#4}\endgroup}
% Table of authorities: a kind's title 13.2 bp above, 8.4 bp below; each entry 7.2 bp above, hanging 0.4 in, the
% page column 40 bp or the width of a longer page list. #1 entry, #2 pages.
\newlength\scotus@w
\newcommand{\toaline}[2]{%
  \begingroup\settowidth{\scotus@w}{#2}\setlength{\xe@col}{40bp}\ifdim\scotus@w>\xe@col\setlength{\xe@col}{\scotus@w}\fi
  \xe@row{7.2bp}{0.4in}{#1}{#2}\endgroup}
\newcommand{\toaheading}[1]{\par\addvspace{13.2bp}\noindent\textbf{#1}\par\nobreak\addvspace{8.4bp}}
% A table's title: centred bold, 12 bp below.
\newcommand{\scotustitle}[1]{\begingroup\centering\bfseries #1\par\endgroup\nobreak\addvspace{12bp}}

% Pages of one authority: #1 its citation labels (a comma list). Each distinct page once, in page order (typst:
% dedup, sorted); six or more pages print "passim" (passim-threshold in the typst template).
\newcounter{toapages}
\newcount\toa@n
\def\toaout{}
\def\toapageone#1{%
  \@ifundefined{r@#1}{}{\expandafter\gdef\csname toa@on@\getpagerefnumber{#1}\endcsname{}}}
\def\toapagesloop#1,{%
  \ifx\relax#1\else
    \toapageone{#1}%
    \expandafter\toapagesloop
  \fi}
\newcommand{\toapages}[1]{%
  \begingroup
  \toapagesloop#1,\relax,%
  \setcounter{toapages}{0}\def\toaout{}\toa@n=0
  \loop\advance\toa@n by 1
    \@ifundefined{toa@on@\the\toa@n}{}{%
      \stepcounter{toapages}%
      \ifx\toaout\empty\edef\toaout{\the\toa@n}\else\edef\toaout{\toaout, \the\toa@n}\fi
      \global\expandafter\let\csname toa@on@\the\toa@n\endcsname\relax}%
  \ifnum\toa@n<999 \repeat
  \ifnum\value{toapages}>5 passim\else\toaout\fi
  \endgroup}
% Zero-width label on a citation: its page is the page of the citation.
\newcommand{\toacite}[1]{\label{#1}}

\newcommand{\scotusline}[2]{\par\addvspace{#1bp}\begingroup\centering #2\par\endgroup}
