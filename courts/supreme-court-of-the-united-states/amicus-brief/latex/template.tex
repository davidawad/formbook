% Headings. Section: centred bold capitals. Points: label in a hanging box; the text hangs under the text.
\newcommand{\scotussection}[1]{%
  \par\addvspace{1.4em}\needspace{4\baselineskip}%
  \begin{center}\textbf{\MakeUppercase{#1}}\end{center}\par\nopagebreak\addvspace{0.2em}\nopagebreak\noindent}
\newcommand{\scotuspoint}[3]{%
  % #1 indent, #2 printed label, #3 text
  \par\addvspace{0.9em}\needspace{3\baselineskip}%
  {\leftskip=#1\relax\hangindent=0.4in\hangafter=1\relax
   \makebox[0.4in][l]{\textbf{#2}}\textbf{#3}\par}\nopagebreak\addvspace{0.3em}\nopagebreak\noindent}

% Pages of one authority: #1 its citation labels (comma list, in document order, so the pages come out in
% order and a repeated page is always the last one printed). Six or more pages print "passim"
% (passim-threshold in the typst template).
\newcounter{toapages}
\def\toalast{-1}
\def\toaout{}
\def\toapageone#1{%
  \@ifundefined{r@#1}{\def\toapg{0}}{\edef\toapg{\getpagerefnumber{#1}}}%
  \ifnum\toapg=\toalast\else
    \ifx\toaout\empty\edef\toaout{\toapg}\else\edef\toaout{\toaout, \toapg}\fi
    \stepcounter{toapages}%
    \xdef\toalast{\toapg}%
  \fi}
\def\toapagesloop#1,{%
  \ifx\relax#1\else
    \toapageone{#1}%
    \expandafter\toapagesloop
  \fi}
\newcommand{\toapages}[1]{%
  \setcounter{toapages}{0}%
  \def\toalast{-1}%
  \def\toaout{}%
  \toapagesloop#1,\relax,%
  \ifnum\value{toapages}>5 passim\else\toaout\fi}
% Zero-width label on a citation: its page is the page of the citation.
\newcommand{\toacite}[1]{\label{#1}}
