% ---------------------------------------------------------------------------
% Table of authorities (typst: toa-section). Pages come from the links in the body (aux file).
% ---------------------------------------------------------------------------
\DeclareRobustCommand{\catoahit}[2]{#2\protected@write\@auxout{}{\string\catoaentry{#1}{\thepage}}}
% Page lists are built in the aux file's order. Each (authority, page) pair is kept once; the page string
% is "p1, p2, ..." and the count is kept beside it (seven or more pages print as "passim", typst passim-threshold).
\def\catoatmp{}
\newcommand{\catoaentry}[2]{%
  \@ifundefined{catoaseen@#1@#2}{%
    \expandafter\gdef\csname catoaseen@#1@#2\endcsname{}%
    \@ifundefined{catoa@#1}{%
      \expandafter\gdef\csname catoa@#1\endcsname{#2}%
      \expandafter\gdef\csname catoan@#1\endcsname{1}%
    }{%
      \expandafter\edef\expandafter\catoatmp\expandafter{\csname catoa@#1\endcsname}%
      \expandafter\xdef\csname catoa@#1\endcsname{\catoatmp, #2}%
      \expandafter\xdef\csname catoan@#1\endcsname{\number\numexpr\csname catoan@#1\endcsname+1\relax}%
    }%
  }{}}
% No list exists until the first pass has written the aux file: then nothing is printed.
\newcommand{\catoapages}[1]{%
  \@ifundefined{catoa@#1}{}{%
    \ifnum\csname catoan@#1\endcsname>6 \emph{passim}\else\csname catoa@#1\endcsname\fi}}
% A group heading; #2 is 1 for the first group, which carries the "Page(s)" label.
\newcommand{\catoagroup}[2]{%
  \par\begingroup\casingle
  \ifnum#2=1 \addvspace{1em}\else \addvspace{1.4em}\fi
  \noindent\textbf{#1}\ifnum#2=1 \hfill\textbf{\underline{Page(s)}}\fi\par\addvspace{0.2em}\endgroup
  \nopagebreak}
% One authority: #1 name (may be empty; a case name is italic), #2 citation, #3 key.
\newcommand{\catoarow}[3]{%
  \par\begingroup\casingle\addvspace{0.7em}\hangindent=0.35in\hangafter=1\relax\noindent
  \ifx\relax#1\relax\else #1,\\\fi
  #2\hspace{0.8em}\cadots\hspace{0.3em}\catoapages{#3}\par\endgroup}
