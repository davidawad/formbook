\newcommand{\ctenwalk}[2]{%
  \ifnum\cteni>#2\relax\else
    \protected@edef\ctencur{\pageref{cten-toa-#1-\the\cteni}}%
    \ifx\ctencur\ctenprev\else
      \ifnum\ctencnt=0 \protected@edef\ctenlist{\ctencur}\else\protected@edef\ctenlist{\ctenlist, \ctencur}\fi
      \advance\ctencnt by 1
      \protected@edef\ctenprev{\ctencur}%
    \fi
    \advance\cteni by 1
    \ctenwalk{#1}{#2}%
  \fi}
