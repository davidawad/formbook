% U.S. Court of Appeals for the Tenth Circuit: appellant's opening brief (LaTeX engine).
%
% Matches the typst template (../typst/ca10-opening-brief.typ) in the mandated form: Times (newtx) 14 pt text and
% footnotes, double spacing (31.4 bp line boxes), one-inch margins, no hyphenation, left alignment. The cover, the
% tables, the statements, the certificates and the signature block are built by ca10-opening-brief.lua from the YAML
% frontmatter; this file holds the page setup and the macros that filter emits. Rule IDs (CA10-...) are in
% ../README.md, "Governing rules". Compiled with stock pdflatex: the fonts are TeX Live packages (no repo-local file,
% no system font).
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by ca10-opening-brief.lua. Every size and space here is in bp (14 bp = typst's 14 pt),
% and blocks are spaced with \addvspace (the larger of the space below one and above the next).
\documentclass[14pt,letterpaper]{extarticle}
% CA10-FORM-002 (FRAP 32(a)(5)(A)): proportionally spaced serif; Times New Roman substitute per docs/FONTS.md.
\usepackage[T1]{fontenc}
% Times: newtxtext/newtxmath ship with every TeX Live, so the generated .tex compiles with stock pdflatex and no
% repo-local file; the faces (TeXGyreTermesX, NewTX) are metric-compatible with Times New Roman.
\usepackage{newtxtext,newtxmath}
% CA10-FORM-003 (FRAP 32(a)(4)): letter, 1 in margins; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=36bp]{geometry}
\usepackage{enumitem}
\usepackage{booktabs,longtable}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
% Links in the brief are citations (toa:) or none; the fallbacks keep pandoc's output valid.
\providecommand{\href}[2]{#2}
\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}

\makeatletter
% Line boxes [typst pitch-double, pitch-single]: CA10-FORM-001 double spacing, 31.4 bp, for the body; single
% spacing, 20 bp, for headings, quotations, footnotes, the TOC/TOA, the cover and the certificates. 14 bp everywhere
% (FRAP 32(a)(5)(A)), footnotes too; section headings 16 bp [heading-size].
\renewcommand\normalsize{\xe@lines{14}{31.4}}
\newcommand\singlespacing{\xe@lines{14}{20}}
\newcommand\headingsize{\xe@lines{16}{20}}
\renewcommand\footnotesize{\xe@lines{14}{20}}
\normalsize
\setlength{\parindent}{0.5in} % after xengine.sty's \raggedright, which sets it to zero
\setcounter{secnumdepth}{-\maxdimen}
\xe@lists{0.5in}
% The page number in the 14 bp type, single-spaced.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\singlespacing\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}
\pagestyle{plain}

\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{#1}

% Footnotes [footnote.entry]: 12.6 bp clear of the text, a 30 % rule, 7 bp above each note; the mark (58 % of the
% note's 14 pt) and a quarter em before the text. LaTeX resets the right skip inside a footnote: it is set back here.
\setlength{\skip\footins}{12.6bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}
\xe@footnotemarks{14}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=7bp\relax
  \vskip 0pt\noindent\xe@mark{\@thefnmark}\hskip 0.25em #1}

% The gaps between the cover's parts: 7 bp, more to fill the page, none when the page is full (typst: the largest
% gap at which the cover fits one page).
\newcommand\cagap{\vskip 7bp plus 1fill minus 7bp}
% A block in the right half of the cover or the signature (typst pad(left: 50%)).
\newcommand\carighthalf{\leftskip=0.5\linewidth\relax\parindent=0pt\relax}

% Block quotations [quote]: 0.5 in in from both margins, single-spaced, 14 bp above and below, 12.6 bp between
% paragraphs.
\renewenvironment{quote}{%
  \par\addvspace{14bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\singlespacing\parindent=0pt\relax
  \parskip=12.6bp\relax
}{%
  \par\endgroup\addvspace{14bp}%
}

% Section headings [style-heading, level 1]: bold capitals, centered, 16 bp; 21 bp above, 15.4 bp below. #1 title,
% #2 label (the TOC refers to ca-hd-<label>). A heading never ends a page (\nobreak, and no orphans after it).
\newcommand{\casection}[2]{%
  \par\addvspace{21bp}%
  \begingroup\centering\headingsize\bfseries\leavevmode\label{ca-hd-#2}\MakeUppercase{#1}\par\endgroup
  \nobreak\addvspace{15.4bp}}
% A point heading of the Argument [level 2]: bold, centered, the roman numeral first; 16.8 bp above, 14 bp below.
\newcommand{\capoint}[3]{%
  \par\addvspace{16.8bp}%
  \begingroup\centering\singlespacing\bfseries\leavevmode\label{ca-hd-#3}#1 #2\par\endgroup
  \nobreak\addvspace{14bp}}
% A numbered sub-heading: the label in a 0.5 in box at #1 half inches, the text hanging after it; 15.4 bp above,
% 12.6 bp below. #2 label, #3 title, #4 TOC label, #5 1 for bold italic (level 4 and below), else bold.
\newcommand{\casub}[5]{%
  \par\addvspace{15.4bp}%
  \begingroup\singlespacing\noindent
  \hangindent=\dimexpr0.5in*(#1+1)\relax\hangafter=1\relax
  \hspace*{\dimexpr0.5in*#1\relax}\label{ca-hd-#4}%
  \ifnum#5=1 \makebox[0.5in][l]{\textbf{\emph{#2}}}\textbf{\emph{#3}}\else
    \makebox[0.5in][l]{\textbf{#2}}\textbf{#3}\fi\par
  \endgroup\nobreak\addvspace{12.6bp}}

% One table-of-contents row [listing-row]: the text wraps before a 36 bp page column, a dot leader runs to the page.
% #1 level, #2 indent in half inches, #3 label, #4 title, #5 TOC label.
\newcommand{\tocline}[5]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}%
  \xe@row{\ifnum#1=1 8.4bp\else 4.9bp\fi}{\dimexpr0.5in*(#2+1)\relax}{%
    \hspace*{\dimexpr0.5in*#2\relax}\ifnum#1>1 \makebox[0.5in][l]{#3}\fi
    \ifnum#1=1 \MakeUppercase{#4}\else#4\fi}{\pageref{ca-hd-#5}}%
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; 7 or more distinct pages print as "passim" (typst passim-threshold).
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority [listing-row]: #1 name (empty for a statute etc.), #2 cite, #3 1 for a case, #4 key; the page column
% is 56 bp, or the width of a longer page list.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{9.8bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
% A group heading. #1 title, #2 1 for the first group (carries the "Page(s)" column head); 14 bp above the first,
% 19.6 bp above the others, 2.8 bp below. It stays with its first entry.
\newcommand{\toagroup}[2]{%
  \par\ifnum#2=1 \addvspace{14bp}\else\addvspace{19.6bp}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi\par\endgroup\nobreak\addvspace{2.8bp}}
% One row of the index of attachments [listing-row]: an 84 bp column for "Attachment N".
\newcommand{\attachrow}[2]{%
  \begingroup\singlespacing\setlength{\xe@col}{84bp}%
  \xe@row{9.8bp}{0.35in}{#1}{Attachment #2}\endgroup}
\makeatother
