% Headings. Section (level 1): bold capitals, centered. Argument point headings (level 2) are bold
% small capitals, centered, with a roman numeral. Elsewhere, and below the point, the labels are

\newif\ifinargument

\newcommand{\BriefLabelText}[1]{%
  \ifnum#1=2 \ifinargument\Roman{briefTwo}.\else\Alph{briefTwo}.\fi\fi
  \ifnum#1=3 \ifinargument\Alph{briefThree}.\else\arabic{briefThree}.\fi\fi
  \ifnum#1=4 \ifinargument\arabic{briefFour}.\else\alph{briefFour}.\fi\fi
  \ifnum#1=5 \ifinargument\alph{briefFive}.\else(\arabic{briefFive})\fi\fi}

% Section heading (level 1): centered, bold capitals, in the table of contents.
\newcommand{\BriefSection}[2]{%
  \setcounter{briefTwo}{0}\setcounter{briefThree}{0}\setcounter{briefFour}{0}\setcounter{briefFive}{0}%
  \BriefMark{1}{#2}\BriefFrontHeading{#1}%
  \addcontentsline{toc}{tocLevelOne}{\uppercase{#1}}}
% Front-matter heading: the same look, not in the table of contents.
\newcommand{\BriefFrontHeading}[1]{%
  \par\Needspace{5\baselineskip}\addvspace{1.5em}
  \begin{BriefSingleSpace}\centering\textbf{\MakeUppercase{#1}}\par\end{BriefSingleSpace}%
  \nopagebreak\vspace{1.1em}\nopagebreak\par\nopagebreak}

% Sub-heading at level n (2 to 5).
\newcommand{\BriefSub}[3]{%
  \BriefMark{#1}{#3}%
  \ifnum#1=2 \stepcounter{briefTwo}\setcounter{briefThree}{0}\setcounter{briefFour}{0}\setcounter{briefFive}{0}\fi
  \ifnum#1=3 \stepcounter{briefThree}\setcounter{briefFour}{0}\setcounter{briefFive}{0}\fi
  \ifnum#1=4 \stepcounter{briefFour}\setcounter{briefFive}{0}\fi
  \ifnum#1=5 \stepcounter{briefFive}\fi
  \edef\BriefLabel{\BriefLabelText{#1}}%
  \ifnum#1=2 \ifinargument
    \par\Needspace{5\baselineskip}\addvspace{1.2em}
    \begin{BriefSingleSpace}\centering\textbf{\BriefLabel\ #2}\par\end{BriefSingleSpace}%
    \nopagebreak\vspace{1em}\nopagebreak\par\nopagebreak
    \edef\BriefEntry{\noexpand\addcontentsline{toc}{tocLevelTwo}{\noexpand\BriefTocEntry{0pt}{\BriefLabel}{#2}}}%
    \BriefEntry
  \else
    \BriefHang{#1}{#2}%
  \fi\else
    \BriefHang{#1}{#2}%
  \fi}
\newcommand{\BriefHang}[2]{%
  \dimen0=\dimexpr\BriefTab*(#1-\ifinargument3\else2\fi+1)\relax
  \par\Needspace{5\baselineskip}\addvspace{1.1em}
  \begin{BriefSingleSpace}%
    \hangindent=\dimexpr\dimen0+\BriefTab\relax\hangafter=1
    \noindent\hspace*{\dimen0}\makebox[\BriefTab][l]{\ifnum#1>3 \textbf{\textit{\BriefLabel}}\else\textbf{\BriefLabel}\fi}%
    \ifnum#1>3 \textbf{\textit{#2}}\else\textbf{#2}\fi\par
  \end{BriefSingleSpace}%
  \nopagebreak\vspace{0.9em}\nopagebreak\par\nopagebreak
  \edef\BriefEntry{\noexpand\addcontentsline{toc}{\BriefTocName{#1}}{\noexpand\BriefTocEntry{\the\dimen0}{\BriefLabel}{#2}}}%
  \BriefEntry}

\newcommand{\toarecord}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1@#2}{%
    \expandafter\gdef\csname toapg@#1@#2\endcsname{}%
    \@ifundefined{toacount@#1}{\expandafter\gdef\csname toacount@#1\endcsname{0}\expandafter\gdef\csname toalist@#1\endcsname{}}{}%
    \expandafter\xdef\csname toalist@#1\endcsname{\ifnum\csname toacount@#1\endcsname=0 \else\csname toalist@#1\endcsname, \fi#2}%
    \expandafter\xdef\csname toacount@#1\endcsname{\number\numexpr\csname toacount@#1\endcsname+1\relax}%
  }{}}
\newcommand{\toapages}[1]{%
  \@ifundefined{toacount@#1}{??}{%
    \ifnum\csname toacount@#1\endcsname>6 \emph{passim}\else\csname toalist@#1\endcsname\fi}}
