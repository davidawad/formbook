% Eighth Circuit reply brief, LaTeX engine (docs/EVAL-LOOP.md, "Engines").
% Mirrors ../typst/ca8-reply-brief.typ: page order, Times (newtx) 14 pt type everywhere, double spacing (28 bp line
% boxes), one-inch margins, no hyphenation, left alignment. ca8.lua builds the cover, the front matter, the tables,
% the signature block and the certificates from the YAML frontmatter; this file holds the page setup and the macros
% that filter emits. Compiled by fb render with stock pdflatex: the fonts are TeX Live packages (no repo-local file,
% no system font).
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by ca8.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks are
% spaced with \addvspace (the larger of the space below one and above the next). The names in [brackets] are the
% typst template's functions.
\documentclass[14pt,letterpaper]{extarticle}
% FRAP 32(a)(5)(A): proportionally spaced serif, 14 point.
\usepackage[T1]{fontenc}
% Times: newtxtext/newtxmath ship with every TeX Live, so the generated .tex compiles with stock pdflatex and no
% repo-local file; the faces (TeXGyreTermesX, NewTX) are metric-compatible with Times New Roman.
\usepackage{newtxtext,newtxmath}
% FRAP 32(a)(4): letter, 1 in margins; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=36bp]{geometry}
\usepackage{enumitem}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
\usepackage{xetimes} % lib/latex/xetimes.sty: newtx at Times New Roman's advances; a space in em is a multiple of \xe@em

\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}

\makeatletter

\setcounter{secnumdepth}{-\maxdimen}

\def\check@icl{}\def\check@icr{}

\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty

% Block quotations [quote]: 0.5 in in from both margins, single-spaced, 14 bp above and below, 12.6 bp between

  \par\endgroup\addvspace{14bp}%

% Section headings (level 1: centered, bold capitals) and numbered headings. #1 level, #2 indent in
% half inches (the label sits at #2 x 0.5 in), #3 label, #4 title, #5 index (the TOC refers to
% ca-hd-<index>), #6 the level from which the text is italic. [style-heading]: 21 / 15.4 bp around a section heading,
% 15.4 / 12.6 bp around a numbered one; a heading never ends a page (\nobreak, and no orphans after it).
\newcommand{\cahead}[6]{%
  \ifnum#1=1
    \par\addvspace{21bp}%
    \begingroup\centering\singlespacing\bfseries\leavevmode\label{ca-hd-#5}\MakeUppercase{#4}\par\endgroup
    \nobreak\addvspace{15.4bp}%
  \else
    \par\addvspace{15.4bp}%
    \begingroup\singlespacing\noindent
    \hangindent=\dimexpr0.5in*(#2+1)\relax\hangafter=1\relax
    \hspace*{\dimexpr0.5in*#2\relax}\label{ca-hd-#5}%
    \makebox[0.5in][l]{\textbf{#3}}% (the label is never italic)
    \ifnum#1<#6 \textbf{#4}\else\textbf{\emph{#4}}\fi\par
    \endgroup\nobreak\addvspace{12.6bp}%
  \fi}

% One table-of-contents row [listing-row]: the text wraps before a 36 bp page column, a dot leader runs to the page.
% #1 level, #2 indent in half inches, #3 label, #4 title, #5 index.
\newcommand{\tocline}[5]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}%
  \xe@row{\ifnum#1=1 8.4bp\else 4.9bp\fi}{\dimexpr0.5in*(#2+1)\relax}{%
    \ifnum#1>1 \hspace*{\dimexpr0.5in*#2\relax}\makebox[0.5in][l]{#3}\fi
    \ifnum#1=1 \MakeUppercase{#4}\else#4\fi}{\pageref{ca-hd-#5}}%
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; 7 or more distinct pages print as "passim" (typst passim-threshold). No \pageref is expanded here, so the page numbers come from the same aux pass that LaTeX already
% reruns for.
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority [listing-row]: #1 name (empty for a statute etc.), #2 cite, #3 1 for a case, #4 key; 7 bp above;
% the page column is 56 bp, or the width of a longer page list.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{7bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
% A group heading: #1 title, #2 1 for the first group (14 bp above, and the underlined "Page(s)" column head at the
% right), 19.6 bp above the others; 2.8 bp below; it stays with its first entry.
\newcommand{\toagroup}[2]{%
  \par\ifnum#2=1 \addvspace{14bp}\else\addvspace{19.6bp}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi\par\endgroup\nobreak\addvspace{2.8bp}}

% A paragraph of a form [par spacing: 14 pt]: 14 bp from the paragraph before it; a block that follows brings its own.
\newcommand\caformpar[1]{\par\addvspace{14bp}\noindent #1\par}
