% The glyphs whose advance differs between the TeX Gyre Schola the typst template and Word set (the vendored OpenType
% fonts/tex-gyre-schola) and tgschola's Type 1 metrics: ( ) [ ] / * (OpenType 483, 452, 447 and 420 thousandths of an
% em, bold 519, 482, 462 and 483; Type 1 333 for the parentheses and brackets, 278 for the slash). Every other glyph
% agrees. ca6-answering-brief.lua writes each of them as \cag{<glyph>}{<em>}{<bold em>}: a box of the OpenType
% advance, so every line holds what typst's holds. The glyph is centred in it with 0.5 bp of white space either side
% and widened to fill the rest: two such glyphs side by side ("(C)(8)") stay 1 bp apart, less than the 1.4 bp
% (a tenth of the type size) at which a PDF text reader takes a gap for a space; a narrower OpenType glyph (the
% asterisk) is narrowed to fit.
\newlength\ca@gw
\newcommand\cag[3]{\begingroup\settowidth\ca@gw{#1}%
  \if b\expandafter\@car\f@series\@nil\dimen@=#3em\relax\else\dimen@=#2em\relax\fi
  \dimen@i=\dimen@\ifdim\dimen@>\dimexpr\ca@gw+1bp\relax\advance\dimen@i by -1bp\fi
  \makebox[\dimen@][c]{\scalebox{\fpeval{\dimen@i/\ca@gw}}[1]{#1}}\endgroup}

% One table-of-contents row [listing-row]: the text wraps before a 36 bp page column, a dot leader runs to the page;
% rows single-spaced with no space between them. #1 indent in half inches (the text hangs half an inch more), #2 the

\newcommand{\tocline}[4]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}\def\ca@lab{#2}%

    \ifx\ca@lab\empty\else\makebox[0.5in][l]{#2}\fi#3}{\pageref{ca-hd-#4}}%
  \endgroup}
\newcommand{\tochead}{\par\addvspace{4.2bp}\begingroup\singlespacing\noindent\hfill\textbf{\underline{Page}}\par\endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time); the next run
% reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in document order; 7 or more
% distinct pages print as "passim" (typst passim-threshold).
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority [listing-row]: #1 name (empty for a statute etc.), #2 cite, #3 1 for a case, #4 key; the page column
% is 56 bp, or the width of a longer page list.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{9.8bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
