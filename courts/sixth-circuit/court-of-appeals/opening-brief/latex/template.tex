% Headings. Labels hang in a 0.5 in box; each level steps in 0.5 in (typst: label-indent, heading-tab).
% Each heading ends with \nopagebreak, so the paragraph after it cannot be left alone at a page foot. The
% TOC entry is written right after the heading box: a write whatsit must not sit between the penalty and the
% glue, because glue after a whatsit is a legal break point.
\newcommand{\caHeadOne}[1]{%
  \Needspace{5\baselineskip}\par\nopagebreak\vspace{21bp}%
  {\singlespacing\centering\fontsize{16.1bp}{19.95bp}\selectfont\bfseries\MakeUppercase{#1}\par}%
  \addcontentsline{toc}{caone}{\protect\caTocLine{}{\protect\MakeUppercase{#1}}}%
  \nopagebreak\vspace{15.4bp}}
\newcommand{\caHeadOneNoToc}[1]{%

  \nopagebreak\vspace{15.4bp}}
\newcommand{\caHeadPoint}[2]{%

% Table of authorities. \toahit records each citation's printed page (written at shipout, so the page is the
% page the citation sits on). \toaref keeps one entry per page; \toapages lists them; "passim" from 7 pages.
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toaref{#1}{\thepage}}}
\newcommand{\toaref}[2]{%
  \@ifundefined{toalast@#1}{\expandafter\gdef\csname toalast@#1\endcsname{}\expandafter\gdef\csname toacount@#1\endcsname{0}}{}%
  \edef\caTmp{#2}%
  \expandafter\ifx\csname toalast@#1\endcsname\caTmp\relax\else
    \expandafter\gdef\csname toalast@#1\endcsname{#2}%
    \expandafter\xdef\csname toacount@#1\endcsname{\number\numexpr\csname toacount@#1\endcsname+1\relax}%
    \@ifundefined{toapages@#1}{\expandafter\gdef\csname toapages@#1\endcsname{#2}}%
      {\expandafter\xdef\csname toapages@#1\endcsname{\csname toapages@#1\endcsname, #2}}%
  \fi}
\newcommand{\caToaPages}[1]{%
  \@ifundefined{toacount@#1}{}{%
    \ifnum\csname toacount@#1\endcsname>6 \emph{passim}\else\csname toapages@#1\endcsname\fi}}
\newcommand{\caToaGroup}[2]{%
  \Needspace{3\baselineskip}\par\vspace{1em}%
  {\singlespacing\noindent\textbf{#1}\ifnum#2=1\hfill\textbf{\underline{Page(s)}}\fi\par}\vspace{0.2em}}
\newcommand{\caToaEntry}[2]{%
  \par\vspace{0.7em}{\singlespacing\noindent\hangindent=0.35in\hangafter=1 #1\hspace{0.8em}%
  \leaders\hbox to 0.16em{.\hss}\hfill\hspace{0.3em}\caToaPages{#2}\par}}
