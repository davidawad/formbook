% Section headings (level 1: centered, bold capitals) and numbered headings. #1 level, #2 indent in
% half inches (the label sits at #2 x 0.5 in), #3 label, #4 title, #5 index (the TOC refers to
% ca-hd-<index>), #6 the level from which the text is italic.
\newcommand{\cahead}[6]{%
  \ifnum#1=1
    \par\needspace{5\baselineskip}\addvspace{1.5em}%
    \begin{center}\singlespacing\textbf{\MakeUppercase{#4}}\label{ca-hd-#5}\par\end{center}%
    \addvspace{1.1em}%
  \else
    \par\needspace{3\baselineskip}\addvspace{1.1em}%
    \begingroup\singlespacing\noindent
    \hangindent=\dimexpr0.5in*(#2+1)\relax\hangafter=1\relax
    \hspace*{\dimexpr0.5in*#2\relax}%
    \makebox[0.5in][l]{\textbf{#3}}%
    \ifnum#1<#6 \textbf{#4}\else\textbf{\emph{#4}}\fi\label{ca-hd-#5}\par
    \endgroup\addvspace{0.9em}%
  \fi}

  \addvspace{1em}}

% One table-of-contents row. #1 level, #2 indent in half inches, #3 label, #4 title, #5 index.
\newcommand{\tocline}[5]{%
  \par\ifnum#1=1 \addvspace{0.6em}\else\addvspace{0.35em}\fi
  \begingroup\singlespacing\noindent
  \ifnum#1=1 \hangindent=0.5in\else\hangindent=\dimexpr0.5in*(#2+1)\relax\fi
  \hangafter=1\relax
  \ifnum#1>1 \hspace*{\dimexpr0.5in*#2\relax}\makebox[0.5in][l]{#3}\fi
  \ifnum#1=1 \MakeUppercase{#4}\else#4\fi
  \hspace{0.8em}\leaders\hbox{.\hspace{0.16em}}\hfill
  \makebox[2.6em][r]{\pageref{ca-hd-#5}}\par
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; 7 or more distinct pages print as "passim" (typst passim-threshold). No \pageref
% is expanded here, so the page numbers come from the same aux pass that LaTeX already reruns for.
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority. #1 name (empty for a constitutional provision etc.), #2 cite, #3 1 for a case, #4 key.
\newcommand{\toaitem}[4]{%
  \par\needspace{3\baselineskip}\addvspace{0.7em}%
  \begingroup\singlespacing\noindent
  \hangindent=0.35in\hangafter=1\relax
  \def\toaname{#1}\ifx\toaname\empty\else\ifnum#3=1 \emph{#1}\else#1\fi,\\\fi
  #2\hspace{0.8em}\leaders\hbox{.\hspace{0.16em}}\hfill\hspace{0.3em}\toapages{#4}\par
  \endgroup}
% A group heading. #1 title, #2 1 for the first group (carries the "Page(s)" column head).
\newcommand{\toagroup}[2]{%
  \par\needspace{3\baselineskip}\ifnum#2=1 \addvspace{1em}\else\addvspace{1.4em}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hfill\textbf{\underline{Page(s)}}\fi\par\endgroup\addvspace{0.2em}}
\makeatother
