% U.S. Court of Appeals for the Seventh Circuit: reply brief (LaTeX engine).
%
% Matches the typst template (../typst/ca7-reply-brief.typ) in the mandated form: Times (newtx) 12 pt
% text, 11 pt footnotes, double spacing (27.7 bp line boxes), one-inch margins, no hyphenation, left
% alignment. The cover, tables, certificates and the signature block are built by ca7-reply-brief.lua
% from the YAML frontmatter; this file holds the page setup and the macros that filter emits.
% Compiled with stock pdflatex: the fonts are TeX Live packages (no repo-local file, no system font).
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by ca7-reply-brief.lua. Every size and space here is in bp (12 bp = typst's 12 pt),
% and blocks are spaced with \addvspace (the larger of the space below one and above the next).
\documentclass[12pt,letterpaper]{extarticle}
% FRAP 32(a)(5)(A) and Cir. R. 32(b): proportionally spaced serif; Times New Roman substitute per docs/FONTS.md.
\usepackage[T1]{fontenc}
% Times: newtxtext/newtxmath ship with every TeX Live, so the generated .tex compiles with stock pdflatex and no
% repo-local file; the faces (TeXGyreTermesX, NewTX) are metric-compatible with Times New Roman.
\usepackage{newtxtext,newtxmath}
% CA7-FORM-003: letter, 1 in margins; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=36bp]{geometry}
\usepackage{enumitem}
\usepackage{booktabs,longtable}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render

\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}

\makeatletter
% Line boxes [typst pitch-double, pitch-single, pitch-note]: CA7-FORM-004 double spacing, 27.7 bp, for the body;
% single spacing, 17 bp, for headings, quotations, the TOC/TOA, the cover, the signature and the certificates;
% Cir. R. 32(b) 11 bp footnotes in 15.5 bp boxes.

\newcommand\singlespacing{\xe@lines{12}{17}}

\setlength{\parindent}{0.5in} % after xengine.sty's \raggedright, which sets it to zero
\setcounter{secnumdepth}{-\maxdimen}
\xe@lists{0.5in}
% The page number in the 12 bp type, single-spaced.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\singlespacing\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}

% Footnotes [footnote.entry]: 7.29 bp clear of the text, a 30 % rule, 6 bp above each note (13.29 bp in all: the room

% text. LaTeX resets the right skip inside a footnote: it is set back here.
\setlength{\skip\footins}{7.29bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}

  \vskip 0pt\noindent\ca@notemark{\@thefnmark}\hskip 0.25em #1}

% Block quotations [quote]: 0.5 in in from both margins, single-spaced, 12 bp above and below, 10.8 bp between
% paragraphs (the filter puts \caquotepar between them).
\newcommand\caquotepar{\par\addvspace{10.8bp}}
\renewenvironment{quote}{%
  \par\addvspace{12bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\singlespacing\parindent=0pt\relax
}{%
  \par\endgroup\addvspace{12bp}%
}

% Section headings (level 1: centered, bold capitals) and numbered headings. #1 level, #2 indent in
% half inches (the label sits at #2 x 0.5 in), #3 label, #4 title, #5 index (the TOC refers to
% ca-hd-<index>), #6 the level from which the text is italic. [style-heading]: 18 / 13.2 bp around a section heading,
% 13.2 / 10.8 bp around a numbered one; a heading never ends a page (\nobreak, and no orphans after it).
\newcommand{\cahead}[6]{%
  \ifnum#1=1
    \par\addvspace{18bp}%
    \begingroup\centering\singlespacing\bfseries\leavevmode\label{ca-hd-#5}\MakeUppercase{#4}\par\endgroup
    \nobreak\addvspace{13.2bp}%
  \else
    \par\addvspace{13.2bp}%
    \begingroup\singlespacing\noindent
    \hangindent=\dimexpr0.5in*(#2+1)\relax\hangafter=1\relax
    \hspace*{\dimexpr0.5in*#2\relax}\label{ca-hd-#5}%
    \makebox[0.5in][l]{\textbf{#3}}%
    \ifnum#1<#6 \textbf{#4}\else\textbf{\emph{#4}}\fi\par
    \endgroup\nobreak\addvspace{10.8bp}%
  \fi}

% One table-of-contents row [listing-row]: the text wraps before a 36 bp page column, a dot leader runs to the page.
% #1 level, #2 indent in half inches, #3 label, #4 title, #5 index.
\newcommand{\tocline}[5]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}%
  \xe@row{\ifnum#1=1 7.2bp\else 4.2bp\fi}{\dimexpr0.5in*(#2+1)\relax}{%
    \ifnum#1>1 \hspace*{\dimexpr0.5in*#2\relax}\makebox[0.5in][l]{#3}\fi
    \ifnum#1=1 \MakeUppercase{#4}\else#4\fi}{\pageref{ca-hd-#5}}%
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; 7 or more distinct pages print as "passim" (typst passim-threshold). No \pageref
% is expanded here, so the page numbers come from the same aux pass that LaTeX already reruns for.
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority [listing-row]: #1 name (empty for a constitutional provision etc.), #2 cite, #3 1 for a case, #4 key;
% the page column is 56 bp, or the width of a longer page list.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{8.4bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
% A group heading. #1 title, #2 1 for the first group (carries the "Page(s)" column head). It stays with its first
% entry.
\newcommand{\toagroup}[2]{%
  \par\ifnum#2=1 \addvspace{12bp}\else\addvspace{16.8bp}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi\par\endgroup\nobreak\addvspace{2.4bp}}
\makeatother
