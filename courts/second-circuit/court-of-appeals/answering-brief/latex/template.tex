% Section headings: bold capitals, centered (typst: 1.15 em, above 1.5 em, below 1.1 em).
\newcommand\briefhead[1]{%
  \par\needspace{2\baselineskip}\addvspace{1.5em}%
  {\centering\briefsingle\bfseries\fontsize{16.1}{19.9}\selectfont\MakeUppercase{#1}\par}%
  \nopagebreak\addvspace{1.1em}\nopagebreak}
% Point headings of the Argument: bold small capitals, centered, the label in front.
\newcommand\briefpoint[2]{%
  \par\needspace{2\baselineskip}\addvspace{1.2em}%
  {\centering\briefsingle\bfseries\scshape #1\ #2\par}%
  \nopagebreak\addvspace{1.1em}\nopagebreak}
% Sub-headings: the label at indent #1 in a half-inch box, the text hanging under it.
% #2 label, #3 style (bold; bold italic from level 4), #4 text.
\newcommand\briefsub[4]{%
  \par\needspace{2\baselineskip}\addvspace{1.1em}%
  \noindent\hspace*{#1}\makebox[\briefheadtab][l]{#3{#2}}%
  \begingroup\parbox[t]{\dimexpr\linewidth-#1-\briefheadtab\relax}{\raggedright #3{#4}}\endgroup
  \par\nopagebreak\addvspace{0.9em}\nopagebreak}

% Table of authorities. A citation writes its page to the .aux file (\toamark); the next run reads the
% marks back (\toaref) and keeps the distinct pages. Seven or more pages print as "passim" (typst).
\newcommand\toamark[1]{\protected@write\@auxout{}{\string\toaref{#1}{\thepage}}}
\newcommand\toaref[2]{%
  \@ifundefined{toacnt@#1}{%
    \expandafter\newcount\csname toacnt@#1\endcsname
    \expandafter\gdef\csname toapg@#1\endcsname{}%
    \expandafter\gdef\csname toalast@#1\endcsname{}}{}%
  \edef\toapgnow{#2}%
  \expandafter\ifx\csname toalast@#1\endcsname\toapgnow\else
    \expandafter\gdef\csname toalast@#1\endcsname{#2}%
    \global\advance\csname toacnt@#1\endcsname 1
    \ifnum\csname toacnt@#1\endcsname>1 \expandafter\g@addto@macro\csname toapg@#1\endcsname{, }\fi
    \expandafter\g@addto@macro\csname toapg@#1\endcsname{#2}%
  \fi}
\newcommand\toapages[1]{%
  \@ifundefined{toacnt@#1}{}{%
    \ifnum\csname toacnt@#1\endcsname>6 \emph{passim}\else\csname toapg@#1\endcsname\fi}}
% One printed authority: key, the lines before the citation (name and line break, or nothing), citation.
\newcommand\toaprint[3]{%
  \@ifundefined{toacnt@#1}{}{%
    \ifnum\csname toacnt@#1\endcsname>0
      \par\addvspace{0.7em}\noindent\begin{minipage}{\linewidth}\briefsingle
      #2\hspace*{0.35in}#3\ \leaders\hbox to 0.16em{\hss.\hss}\hfill\ \toapages{#1}
      \end{minipage}\par
    \fi}}
\newcommand\toaitem[3]{\toaprint{#1}{#2,\\}{#3}}     % key, name, citation
\newcommand\toaitemnoname[2]{\toaprint{#1}{}{#2}}    % key, citation
\newcommand\toagroup[1]{%
  \par\addvspace{1.4em}\noindent\begin{minipage}{\linewidth}\briefsingle\bfseries #1\end{minipage}\par\nopagebreak}
\newcommand\toagroupfirst[1]{%
  \par\addvspace{1em}\noindent\begin{minipage}{\linewidth}\briefsingle\bfseries #1\hfill\underline{Page(s)}\end{minipage}\par\nopagebreak}
