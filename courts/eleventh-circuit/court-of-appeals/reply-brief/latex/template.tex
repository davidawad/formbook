% Table of citations (typst: toa-section). Pages come from the links in the body (aux file), every page is listed.

% A link writes its page and its primary flag (1: the party primarily relies on it) to the aux file.
\DeclareRobustCommand{\catoahit}[3]{#3\protected@write\@auxout{}{\string\catoaentry{#1}{\thepage}{#2}}}
% Each (authority, page) pair is kept once, in document order; a primary page is marked with an asterisk.
\newcommand{\catoaentry}[3]{%
  \@ifundefined{catoaseen@#1@#2}{%
    \expandafter\gdef\csname catoaseen@#1@#2\endcsname{}%
    \@ifundefined{catoa@#1}{%
      \expandafter\gdef\csname catoa@#1\endcsname{#2}%
    }{%
      \expandafter\edef\expandafter\catoatmp\expandafter{\csname catoa@#1\endcsname}%
      \expandafter\xdef\csname catoa@#1\endcsname{\catoatmp,#2}%
    }%
  }{}%
  \ifnum#3=1 \expandafter\gdef\csname catoastar@#1@#2\endcsname{}\fi}
% The page list of one authority: "4, 5, *6", from the first pass's aux file (empty before it exists).
% \catoacur is the authority being listed; \forcsvlist (etoolbox) runs \catoaadd over its pages.
\newcommand{\catoaadd}[1]{%
  \ifx\catoaout\@empty\else\g@addto@macro\catoaout{, }\fi
  \@ifundefined{catoastar@\catoacur @#1}{}{\g@addto@macro\catoaout{*}}%
  \g@addto@macro\catoaout{#1}}
\newcommand{\catoapages}[1]{%
  \@ifundefined{catoa@#1}{}{%
    \begingroup
    \gdef\catoacur{#1}\gdef\catoaout{}%
    \edef\catoalist{\csname catoa@#1\endcsname}%
    \forcsvlist{\catoaadd}{\catoalist}%
    \catoaout
    \endgroup}}
% A group heading; #2 is 1 for the first group, which carries the "Page(s)" label.
\newcommand{\catoagroup}[2]{%
  \par\begingroup\cabody
  \ifnum#2=1 \addvspace{0.6em}\else \addvspace{1.4em}\fi
  \noindent\textbf{#1}\ifnum#2=1 \hfill\textbf{\underline{Page(s)}}\fi\par\endgroup
  \nopagebreak}
% One authority: #1 name (may be empty; a case name is italic), #2 citation, #3 key.

  \ifx#1\catoaempty\else #1,\\\fi
  #2\hspace{0.8em}\cadots\hspace{0.3em}\catoapages{#3}\par\end{samepage}\endgroup}
