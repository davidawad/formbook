% U.S. Court of Appeals for the Eleventh Circuit: appellant's opening brief (FRAP 28, 32; 11th Cir. R. 28-1, 32-3).
% LaTeX engine. The same form as typst/ca11-opening-brief.typ: rule-driven and content-agnostic. Everything printed
% comes from the markdown frontmatter, the body, or latex/ca11.lua (heading labels, table of citations, word count).
% Built by tools/render.sh --engine latex with stock pdflatex; Times from newtx (below). See ../README.md.
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by ca11.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks
% are spaced with \addvspace (the larger of the space below one and above the next).

\usepackage[T1]{fontenc}
% Times: newtxtext/newtxmath ship with every TeX Live, so the generated .tex compiles with stock pdflatex and no
% repo-local file; the faces (TeXGyreTermesX, NewTX) are metric-compatible with Times New Roman (docs/FONTS.md).
\usepackage{newtxtext,newtxmath}
% FRAP 32(a)(4): letter, 1 in margins; the page number in the bottom margin, where typst sets it.

% ---------------------------------------------------------------------------
% Table of citations (typst: toa-section). Pages come from the links in the body (aux file), every page is listed.
% ---------------------------------------------------------------------------
% A link writes its page and its primary flag (1: the party primarily relies on it) to the aux file, from right after
% its first word (ca11.lua puts \catoahit there).
\DeclareRobustCommand{\catoahit}[2]{\protected@write\@auxout{}{\string\catoaentry{#1}{\thepage}{#2}}}
% Each (authority, page) pair is kept once, in document order; a primary page is marked with an asterisk.
\newcommand{\catoaentry}[3]{%
  \@ifundefined{catoaseen@#1@#2}{%
    \expandafter\gdef\csname catoaseen@#1@#2\endcsname{}%
    \@ifundefined{catoa@#1}{%
      \expandafter\gdef\csname catoa@#1\endcsname{#2}%
    }{%
      \expandafter\edef\expandafter\catoatmp\expandafter{\csname catoa@#1\endcsname}%
      \expandafter\xdef\csname catoa@#1\endcsname{\catoatmp,#2}%
    }%
  }{}%
  \ifnum#3=1 \expandafter\gdef\csname catoastar@#1@#2\endcsname{}\fi}
% The page list of one authority: "4, 5, *6", from the first pass's aux file (empty before it exists).
% \catoacur is the authority being listed; \forcsvlist (etoolbox) runs \catoaadd over its pages.
\newcommand{\catoaadd}[1]{%
  \ifx\catoaout\@empty\else\g@addto@macro\catoaout{, }\fi
  \@ifundefined{catoastar@\catoacur @#1}{}{\g@addto@macro\catoaout{*}}%
  \g@addto@macro\catoaout{#1}}
\newcommand{\catoapages}[1]{%
  \@ifundefined{catoa@#1}{}{%
    \begingroup
    \gdef\catoacur{#1}\gdef\catoaout{}%
    \edef\catoalist{\csname catoa@#1\endcsname}%
    \expandafter\forcsvlist\expandafter\catoaadd\expandafter{\catoalist}% (the list itself, not its name)
    \catoaout
    \endgroup}}
% A group heading, single-spaced, 14 bp (the first) or 19.6 bp above, 2.8 bp below; #2 is 1 for the first group,
% which carries the "Page(s)" label. It stays with its first entry.
\newcommand{\catoagroup}[2]{%
  \par\ifnum#2=1 \addvspace{14bp}\else\addvspace{19.6bp}\fi
  \begingroup\casingle\noindent\textbf{#1}\ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi
  \par\endgroup\nobreak\addvspace{2.8bp}}
% One authority [listing-row], double-spaced, 9.8 bp above: #1 name (may be empty; a case name is italic),
% #2 citation, #3 key. The page column is 56 bp, or the width of a longer page list.
\def\catoaempty{}
\newlength\ca@w
\newcommand{\catoarow}[3]{%
  \begingroup\cabody
  \settowidth{\ca@w}{\catoapages{#3}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\catoaname{#1}%
  \xe@row{9.8bp}{0.35in}{\ifx\catoaname\catoaempty\else #1,\newline\fi #2}{\catoapages{#3}}%
  \endgroup}
