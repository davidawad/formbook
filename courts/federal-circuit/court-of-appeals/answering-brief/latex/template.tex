\newcommand\ca@hone[1]{%

\newcommand\ca@htwo[1]{%
  \stepcounter{catwo}%

  \else
    \ca@subhead{\ca@half}{\ca@half}{\ca@labelof{2}}{\textbf{#1}}{\ca@tocwrite{toc}{tocOTwo}{\ca@tmp}}%
  \fi}
\newcommand\ca@hthree[1]{%
  \stepcounter{cathree}%

  {
    \seq_if_exist:cF { g_ca_toa_#1_seq } { \seq_new:c { g_ca_toa_#1_seq } }
    \seq_if_in:cnF { g_ca_toa_#1_seq } { #2 }
      { \seq_gput_right:cn { g_ca_toa_#1_seq } { #2 } }
  }
\cs_new:Npn \ca@toapages #1
  {
    \seq_if_exist:cTF { g_ca_toa_#1_seq }
      {
        \int_compare:nNnTF { \seq_count:c { g_ca_toa_#1_seq } } > { 6 }
          { \emph{passim} }
          { \seq_use:cn { g_ca_toa_#1_seq } { \ca@pagesep } }
      }
      { }
  }
