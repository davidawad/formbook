% U.S. Court of Appeals for the Federal Circuit: reply brief, LaTeX engine.
%
% Rule-driven and content-agnostic, the same layout as ../typst/cafc-reply-brief.typ (the reference for this
% court): Times (newtx) 14 pt everywhere, double spacing (32.2 bp line boxes) in the text, single spacing (16.1 bp)
% for headings, the cover, the tables, footnotes, quotations, the signature and the certificates; one-inch margins;
% no hyphenation; left alignment. No case-specific text, no coordinates measured off any filed brief, no forced line
% breaks. The cover, the tables, the signature block and the certificates are built by cafc.lua from the YAML
% frontmatter; this file holds the page setup and the macros that filter emits. Compiled with stock pdflatex.
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by cafc.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks
% are spaced with \addvspace (the larger of the space below one and above the next).
\documentclass[letterpaper,12pt]{article}
% FRAP 32(a)(5)(A): proportionally spaced serif, 14 point. newtxtext/newtxmath ship with every TeX Live, so the
% generated .tex compiles with stock pdflatex; the faces are metric-compatible with Times New Roman (docs/FONTS.md).
\usepackage[T1]{fontenc}
\usepackage{newtxtext,newtxmath}
% FRAP 32(a)(4): letter, 1 in margins; the page number in the bottom margin, as typst sets it (footer-descent 30 %).
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=34.5bp]{geometry}
\usepackage{enumitem}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
% Links in the brief are citations (toa:) or none; the fallbacks keep pandoc's output valid.
\providecommand{\href}[2]{#2}
\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}

\makeatletter
% Line boxes [typst pitch-double, pitch-single]: FRAP 32(a)(4) double spacing, 32.2 bp, for the text; single
% spacing, 16.1 bp, for everything else. Footnotes are 14 pt too (FRAP 32(a)(5)), single-spaced.
\renewcommand\normalsize{\xe@lines{14}{32.2}}
\newcommand\singlespacing{\xe@lines{14}{16.1}}
\renewcommand\footnotesize{\xe@lines{14}{16.1}}
\normalsize
\setlength{\parindent}{0.5in} % after xengine.sty's \raggedright, which sets it to zero
\setcounter{secnumdepth}{-\maxdimen}
\xe@lists{0.5in}
% The page number in the 14 bp type, single-spaced.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\singlespacing\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}
\pagestyle{plain}

\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{#1}

% Footnotes [footnote.entry]: 12.6 bp clear of the text, a 30 % rule, 7 bp above each note; the mark (58 % of the
% 14 pt type) and a quarter em before the text. LaTeX resets the right skip inside a footnote: it is set back here.
\setlength{\skip\footins}{12.6bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}
\xe@footnotemarks{14}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=7bp\relax
  \vskip 0pt\noindent\xe@mark{\@thefnmark}\hskip 0.25em #1}

% The cover's gaps [cover-body, gap]: 14 bp (1 em) between its parts, less when the page is full (typst: the
% largest gap at which the cover fits one page). #1: the share of the gap (0.6 at the top, 0.3 around "v.").
\newcommand\cagap[1][1]{\par\vskip \fpeval{14*(#1)}bp minus \fpeval{14*(#1)}bp\relax}
% The right half of the cover and of the signature blocks: the text starts at the middle of the line.
\newcommand\carighthalf{\leftskip=0.5\linewidth\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parindent=0pt\relax}

% An ellipsis [typst: the one glyph U+2026, 1 em wide]: three periods spread over 1 em, which the PDF text reads as
% the one character (ActualText), as typst's and Word's glyph.
\newcommand\caellipsis{\pdfliteral page{/Span<</ActualText<FEFF2026>>>BDC}\hbox to 1em{.\hss.\hss.}\pdfliteral page{EMC}}

% A row of the cover's counsel columns: the line keeps the height of its minipages (each sets its own lines).
\newcommand\carow[1]{{\xe@natural\noindent#1\par}}

% Block quotations [quote]: 0.5 in in from both margins, single-spaced, 14 bp above and below, 12.6 bp between
% paragraphs.
\renewenvironment{quote}{%
  \par\addvspace{14bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\singlespacing\parindent=0pt\relax
  \parskip=12.6bp\relax\vskip-\parskip % the first paragraph takes no \parskip: 14 bp above, as typst
}{%
  \par\endgroup\addvspace{14bp}%
}

% A section heading of the front matter and the certificates [style-heading, unnumbered]: centered, bold capitals,
% 21 bp above and 15.4 bp below; a heading never ends a page (\nobreak, and no orphans after it).
\newcommand{\casection}[1]{%
  \par\addvspace{21bp}%
  \begingroup\centering\singlespacing\bfseries\leavevmode\MakeUppercase{#1}\par\endgroup
  \nobreak\addvspace{15.4bp}}
% A numbered section [style-heading, numbered, level 1]: centered, bold, the label and the title in capitals, 21 bp
% above and 15.4 bp below. #1 label, #2 title, #3 index (the contents refer to ca-hd-<index>).
\newcommand{\casectionnum}[3]{%
  \par\addvspace{21bp}%
  \begingroup\centering\singlespacing\bfseries\leavevmode\label{ca-hd-#3}#1 \MakeUppercase{#2}\par\endgroup
  \nobreak\addvspace{15.4bp}}
% A numbered sub-heading [style-heading, numbered]: the label in a half-inch box at #2 half inches, the text hanging
% after it; 15.4 bp above and 12.6 bp below. #1 level (4 and 5: label and text italic), #2 indent in half inches,
% #3 label, #4 title, #5 index.
\newcommand{\cahead}[5]{%
  \par\addvspace{15.4bp}%
  \begingroup\singlespacing\noindent
  \hangindent=\dimexpr0.5in*(#2+1)\relax\hangafter=1\relax
  \hspace*{\dimexpr0.5in*#2\relax}\label{ca-hd-#5}%
  \ifnum#1<4 \makebox[0.5in][l]{\textbf{#3}}\textbf{#4}\else\makebox[0.5in][l]{\textbf{\emph{#3}}}\textbf{\emph{#4}}\fi\par
  \endgroup\nobreak\addvspace{12.6bp}}
% A check box [checkbox]: a ruled box (0.6 bp) around an "X", or around three spaces' width when not checked.
\newcommand\cabox[1]{{\fboxsep=1bp\fboxrule=0.6bp\fbox{\ifnum#1=1 X\else\makebox[3\fontdimen2\font]{\vphantom{X}}\fi}}}

% One table-of-contents row [toc-entry, listing-row]: the label in a half-inch box at #1 half inches, the title
% hanging after it, wrapping before a 36 bp page column; a dot leader runs to the page. #2 label, #3 title, #4 index,
% #5 the space above (8.4 bp before a level-1 row, 4.9 bp before the others).
\newcommand{\tocline}[5]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}%
  \xe@row{#5}{\dimexpr0.5in*(#1+1)\relax}{%
    \hspace*{\dimexpr0.5in*#1\relax}\makebox[0.5in][l]{#2}#3}{\pageref{ca-hd-#4}}%
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; every page is listed ("passim" is not used: proposed Fed. Cir. R. 32(j)).
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \toapgs}
% One authority [listing-row]: #1 name (empty for a statute etc.), #2 cite, #3 1 for a case, #4 key; the page
% column is 56 bp, or the width of a longer page list; wrapped lines hang 0.35 in; 9.8 bp above.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{9.8bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
% A group heading: 14 bp above the first (which carries the "Page(s)" column head), 19.6 bp above the others,
% 2.8 bp below; it stays with its first entry. #1 title, #2 1 for the first group.
\newcommand{\toagroup}[2]{%
  \par\ifnum#2=1 \addvspace{14bp}\else\addvspace{19.6bp}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi\par\endgroup\nobreak\addvspace{2.8bp}}
\makeatother
