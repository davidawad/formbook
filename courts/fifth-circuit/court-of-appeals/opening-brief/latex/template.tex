% ---- headings (ca5-brief.lua computes the labels; these set the style and the TOC entry) ----
% Front headings (certificate of interested persons, oral argument statement): not in the TOC.
\newcommand{\brfront}[1]{\par\begingroup\sglspace\begin{center}\bfseries\MakeUppercase{#1}\end{center}\par\endgroup\nopagebreak\vspace{0.6em}\par\noindent}
% Level 1: centered, bold capitals, in the TOC without a label.
\newcommand{\brone}[1]{\par\vspace{1.1em}\begingroup\sglspace\begin{center}\bfseries\MakeUppercase{#1}\end{center}\par\endgroup
  \addcontentsline{toc}{brone}{\protect\MakeUppercase{#1}}\nopagebreak\vspace{0.6em}\par\noindent}
% Argument point heading (level 2 in the Argument): centered, bold small capitals, roman label.
\newcommand{\brargtwo}[2]{\par\vspace{1.1em}\begingroup\sglspace\begin{center}\bfseries\scshape #1\ #2\end{center}\par\endgroup
  \addcontentsline{toc}{brargtwo}{\protect\numberline{#1}#2}\nopagebreak\vspace{0.6em}\par\noindent}
% Sub-headings: label in a 0.5 in box at indent #1, title hangs under itself. #2 label, #3 title, #4 font switch.
\newcommand{\brsubgen}[4]{\par\vspace{1em}\begingroup\sglspace\noindent
  \hangindent=\dimexpr #1+0.5in\relax\hangafter=1 \hspace*{#1}\makebox[0.5in][l]{#4 #2}#4 #3\par\endgroup
  \addcontentsline{toc}{brsub}{\protect\hspace*{#1}\protect\numberline{#2}#3}\nopagebreak\vspace{0.6em}\par\noindent}
\newcommand{\brsub}[3]{\brsubgen{#1}{#2}{#3}{\bfseries}}
\newcommand{\brsubi}[3]{\brsubgen{#1}{#2}{#3}{\bfseries\itshape}}

% ---- table of authorities ----
% \toacite{key}{text} marks a citation: its page goes to the aux file; the next run reads it back.
% A key cited on passim-threshold pages or more prints "passim" (7, CA5 convention in the typst template).
\newcommand{\toaref}[2]{\@ifundefined{toalast@#1}{%
    \expandafter\gdef\csname toalast@#1\endcsname{#2}%
    \expandafter\gdef\csname toa@#1\endcsname{#2}%
    \expandafter\newcount\csname toan@#1\endcsname
    \global\advance\csname toan@#1\endcsname by 1 }{%
  \ifnum\csname toalast@#1\endcsname=#2 \else
    \expandafter\gdef\csname toalast@#1\endcsname{#2}%
    \expandafter\xdef\csname toa@#1\endcsname{\csname toa@#1\endcsname, #2}%
    \global\advance\csname toan@#1\endcsname by 1 \fi}}
\newcommand{\toacite}[2]{#2\protected@write\@auxout{}{\string\toaref{#1}{\thepage}}}
\newcommand{\toapages}[1]{\@ifundefined{toan@#1}{}{%
  \ifnum\csname toan@#1\endcsname>6 \emph{passim}\else\csname toa@#1\endcsname\fi}}
\newcommand{\brtoagroup}[3]{\par\vspace{#2}\noindent\textbf{#1}\hfill\textbf{\underline{#3}}\par\nopagebreak}
\newcommand{\brtoaentry}[2]{\par\begingroup\sglspace\noindent\hangindent=0.35in\hangafter=1
  #1\ \dotfill\ \toapages{#2}\par\endgroup\vspace{0.7em}}
\makeatother
