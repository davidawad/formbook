% U.S. Court of Appeals for the Fifth Circuit: appellee's (answering) brief, LaTeX engine.
%
% Matches the typst template (../typst/ca5-answering-brief.typ) in the mandated form: Times (newtx, metric-compatible
% with Times New Roman) 14 pt text and footnotes (FRAP 32(a)(5)(A)), double spacing (31.4 bp line boxes), one-inch
% margins, no hyphenation, left alignment. The cover, tables, front statements, signature block and certificates are
% built by ca5-answering-brief.lua from the YAML frontmatter; this file holds the page setup and the macros that filter
% emits. Compiled with stock pdflatex: the fonts are TeX Live packages (no repo-local file, no system font).
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty, which fb render
% inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are lib/latex/xengine.lua, run by
% ca5-answering-brief.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks are spaced with
% \addvspace (the larger of the space below one and above the next).
\documentclass[14pt,letterpaper]{extarticle}
\usepackage[T1]{fontenc}
\usepackage{newtxtext,newtxmath}
% CA5-FORM-002: letter, 1 in margins; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=36bp]{geometry}
\usepackage{enumitem}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
% Links in the brief are citations (toa:) or none; the fallbacks keep pandoc's output valid.
\providecommand{\href}[2]{#2}
\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}
\providecommand{\tightlist}{}
\providecommand{\pandocbounded}[1]{#1}
% The ellipsis character (U+2026) as the font's own glyph, one em wide like Times New Roman's (typst and Word set the
% character; \ldots sets three spaced periods): T1 has no ellipsis, LY1 has it at 133 (newtx ships LY1 fonts; the
% encoding is declared bare, as TeX Live's basic scheme has no ly1enc.def).
\DeclareFontEncoding{LY1}{}{}
\DeclareFontSubstitution{LY1}{ntxtlf}{m}{n}
\newcommand\caellipsis{{\fontencoding{LY1}\selectfont\char133}}

\makeatletter
% Line boxes [typst pitch-double, pitch-single]: CA5-FORM-003 double spacing, 31.4 bp, for the body; single spacing,
% 20 bp, for headings, quotations, footnotes (14 bp, FRAP 32(a)(5)(A)), the TOC/TOA, the cover and the certificates;
% section titles at 16 bp.
\renewcommand\normalsize{\xe@lines{14}{31.4}}
\newcommand\singlespacing{\xe@lines{14}{20}}
\renewcommand\footnotesize{\xe@lines{14}{20}}
\newcommand\titlesize{\xe@lines{16}{20}}
\normalsize
\setlength{\parindent}{0.5in} % after xengine.sty's \raggedright, which sets it to zero
\setcounter{secnumdepth}{-\maxdimen}
\xe@lists{0.5in}
% Widows and orphans are allowed, as in the typst template: typst's widow control moves a line to the next page but
% leaves that line's footnote behind (a note on the page before its reference), so the template turns it off.
\clubpenalty=0
\widowpenalty=0
\displaywidowpenalty=0
% The page number in the 14 bp type, single-spaced.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\singlespacing\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}
\pagestyle{plain}

% Footnotes [footnote.entry]: a 30 % rule 12.6 bp clear of the text, 7 bp between notes (each note opens with its 7 bp
% \parskip, so \skip\footins is the other 5.6 bp of the clearance), never split; the mark (58 % of 14 pt) and a
% quarter em before the text. LaTeX resets the right skip inside a footnote: it is set back here.
\setlength{\skip\footins}{5.6bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}
\xe@footnotemarks{14}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=7bp\relax
  \vskip 0pt\noindent\xe@mark{\@thefnmark}\hskip 0.25em #1}

% The cover: rules in vertical mode with no interline glue (they cost their own height, as in typst); the gaps between
% its parts 7 bp, more to fill the page, less when the page is full (typst: the largest gap at which the cover fits).
\newcommand\cagap{\vskip 7bp plus 1fill minus 7bp}
\newcommand\cadouble{\par\nointerlineskip\hbox to\linewidth{\rule{\linewidth}{1.6pt}}\nointerlineskip\vspace{2.5pt}%
  \nointerlineskip\hbox to\linewidth{\rule{\linewidth}{0.5pt}}\nointerlineskip}
\newcommand\cashort{\par\nointerlineskip\hbox to\linewidth{\hss\rule{2in}{0.7pt}\hss}\nointerlineskip}

% Block quotations [quote]: 0.5 in in from both margins, single-spaced, 14 bp above and below, 12.6 bp between
% paragraphs.
\renewenvironment{quote}{%
  \par\addvspace{14bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\singlespacing\parindent=0pt\relax
  \parskip=12.6bp\relax\vskip-12.6bp\relax % the first paragraph's \parskip: 14 bp above it, not 26.6
}{%
  \par\endgroup\addvspace{14bp}%
}

% Headings [style-heading]. #1 level, #2 indent in half inches (the label sits at #2 x 0.5 in), #3 label, #4 title,
% #5 index (the TOC refers to ca-hd-<index>). Level 1: centered bold capitals at 16 bp, 21 / 15.4 bp around it;
% numbered: 15.4 / 12.6 bp, bold, bold italic from level 4. A heading never ends a page (\nobreak).
\newcommand{\cahead}[5]{%
  \ifnum#1=1
    \par\addvspace{21bp}%
    \begingroup\centering\titlesize\bfseries\leavevmode\label{ca-hd-#5}\MakeUppercase{#4}\par\endgroup
    \nobreak\addvspace{15.4bp}%
  \else
    \par\addvspace{15.4bp}%
    \begingroup\singlespacing\noindent
    \hangindent=\dimexpr0.5in*(#2+1)\relax\hangafter=1\relax
    \hspace*{\dimexpr0.5in*#2\relax}\label{ca-hd-#5}%
    \ifnum#1<4 \makebox[0.5in][l]{\textbf{#3}}\textbf{#4}\else\makebox[0.5in][l]{\textbf{\emph{#3}}}\textbf{\emph{#4}}\fi
    \par\endgroup\nobreak\addvspace{12.6bp}%
  \fi}
% The Argument's point heading: centered, bold, the roman numeral first; 16.8 / 14 bp around it.
\newcommand{\cacenter}[3]{%
  \par\addvspace{16.8bp}%
  \begingroup\centering\singlespacing\bfseries\leavevmode\label{ca-hd-#3}#1 #2\par\endgroup
  \nobreak\addvspace{14bp}}

% One table-of-contents row [listing-row]: the text wraps before a 36 bp page column, a dot leader runs to the page.
% #1 level, #2 indent in half inches, #3 label, #4 title, #5 index.
\newcommand{\tocline}[5]{%
  \begingroup\singlespacing\setlength{\xe@col}{36bp}%
  \xe@row{\ifnum#1=1 8.4bp\else 4.9bp\fi}{\dimexpr0.5in*(#2+1)\relax}{%
    \hspace*{\dimexpr0.5in*#2\relax}\ifnum#1>1 \makebox[0.5in][l]{#3}\fi
    \ifnum#1=1 \MakeUppercase{#4}\else#4\fi}{\pageref{ca-hd-#5}}%
  \endgroup}

% Table of authorities. Each citation writes its page to the .aux file (\toahit, at shipout time);
% the next run reads them back as \toapg@<key> = "p1,p2,...". \toapages prints the distinct pages in
% document order; 7 or more distinct pages print as "passim" (typst passim-threshold).
\newcommand{\toahit}[1]{\protected@write\@auxout{}{\string\toarec{#1}{\thepage}}}
\newcommand{\toarec}[2]{%
  \@ifundefined{toapg@#1}%
    {\expandafter\gdef\csname toapg@#1\endcsname{#2,}}%
    {\expandafter\xdef\csname toapg@#1\endcsname{\csname toapg@#1\endcsname#2,}}}
\newcounter{toacnt}
\newcommand{\toapages}[1]{%
  \setcounter{toacnt}{0}\gdef\toapgs{}\def\toalast{}%
  \@ifundefined{toapg@#1}{}{%
    \expandafter\let\expandafter\toalist\csname toapg@#1\endcsname
    \@for\toap:=\toalist\do{%
      \ifx\toap\empty\else
        \ifx\toap\toalast\else
          \ifnum\value{toacnt}>0 \xdef\toapgs{\toapgs, }\fi
          \xdef\toapgs{\toapgs\toap}\stepcounter{toacnt}%
        \fi
        \edef\toalast{\toap}%
      \fi
    }%
  }%
  \ifnum\value{toacnt}>6 \emph{passim}\else\toapgs\fi}
% One authority [listing-row]: #1 name (empty for a rule etc.), #2 cite, #3 1 for a case, #4 key;
% the page column is 56 bp, or the width of a longer page list.
\newlength\ca@w
\newcommand{\toaitem}[4]{%
  \begingroup\singlespacing
  \settowidth{\ca@w}{\toapages{#4}}%
  \setlength{\xe@col}{56bp}\ifdim\ca@w>\xe@col\setlength{\xe@col}{\ca@w}\fi
  \def\toaname{#1}%
  \xe@row{9.8bp}{0.35in}{\ifx\toaname\empty\else\ifnum#3=1 \emph{#1},\else#1,\fi\newline\fi #2}{\toapages{#4}}%
  \endgroup}
% A group heading. #1 title, #2 1 for the first group (carries the "Page(s)" column head). It stays with its first
% entry.
\newcommand{\toagroup}[2]{%
  \par\ifnum#2=1 \addvspace{14bp}\else\addvspace{19.6bp}\fi
  \begingroup\singlespacing\noindent\textbf{#1}%
  \ifnum#2=1 \hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\fi\par\endgroup\nobreak\addvspace{2.8bp}}
\makeatother
