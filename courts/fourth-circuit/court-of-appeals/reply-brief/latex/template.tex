% Fonts: Times from newtxtext/newtxmath, which ship with every TeX Live, so the generated .tex compiles with stock

% ---- headings (called by ca4.lua) ----
% \brfsection{display}: a level-1 heading: centred, bold, capitals.
\newcommand\cvrule[2]{\par\nointerlineskip\hbox to\linewidth{\hfil\vrule width #1 height #2 depth 0pt\hfil}\par\nointerlineskip}
\newcommand\brfsection[1]{%
  \par\needspace{100pt}\vspace{21pt}%
  {\singlespacing\centering\textbf{#1}\par}%
  \vspace{4.5pt}\nopagebreak}
% \brfsub{indent}{label}{title}: a numbered sub-heading; the label sits at the indent, the title hangs under itself.
\newcommand\brfsub[3]{%
  \par\needspace{70pt}\vspace{15.4pt}%
  {\singlespacing\noindent\hangindent=\dimexpr#1+0.5in\relax\hangafter=1\relax
    \hspace*{#1}\makebox[0.5in][l]{#2}#3\par}%
  \vspace{9pt}\nopagebreak}
% \brfunnum{indent}{title}: an unnumbered sub-heading: bold, left, single-spaced.
\newcommand\brfunnum[2]{%
  \par\needspace{70pt}\vspace{15.4pt}%
  {\singlespacing\noindent\hangindent=#1\relax\hangafter=1\relax\hspace*{#1}\textbf{#2}\par}%
  \vspace{9pt}\nopagebreak}
% \brfpoint{title}: a centred point heading (answering brief: see CFG in ca4.lua).
\newcommand\brfpoint[1]{%
  \par\needspace{70pt}\vspace{15.4pt}%
  {\singlespacing\centering\textbf{#1}\par}%
  \vspace{11pt}\nopagebreak}

% ---- table of authorities: the pages come from the aux file, written by \brfhit after each citation ----
\newcommand\brfhit[1]{\protected@write\@auxout{}{\string\brfcite{#1}{\thepage}}}
% \brfcite{key}{page}: keeps the distinct pages of each key in order, and their count.
\newcommand\brfcite[2]{%
  \@ifundefined{brfp@#1}{%
    \expandafter\gdef\csname brfp@#1\endcsname{#2}%
    \expandafter\gdef\csname brfn@#1\endcsname{1}%
    \expandafter\gdef\csname brfl@#1\endcsname{#2}}{%
    \edef\brf@page{#2}%
    \expandafter\ifx\csname brfl@#1\endcsname\brf@page\else
      \expandafter\xdef\csname brfp@#1\endcsname{\csname brfp@#1\endcsname, #2}%
      \expandafter\xdef\csname brfn@#1\endcsname{\number\numexpr\csname brfn@#1\endcsname+1\relax}%
      \expandafter\gdef\csname brfl@#1\endcsname{#2}%
    \fi}}
% \brfpages{key}: the page list, or "passim" from 7 distinct pages on (typst: passim-threshold).
\newcommand\brfpages[1]{%
  \@ifundefined{brfn@#1}{}{%
    \ifnum\csname brfn@#1\endcsname>6 \emph{passim}\else\csname brfp@#1\endcsname\fi}}
% \brftoaentry{name}{cite}{key}{has name}: the name (for cases, italic) and the cite, then the pages at the right.
\newcommand\brftoaentry[4]{%
  \par\noindent\ifnum#4=1 \needspace{42pt}\else\needspace{26pt}\fi\vspace{5.5pt}%
  {\singlespacing\hangindent=0.35in\hangafter=1\relax
    \ifnum#4=1 #1,\\\fi #2\ %
    \leaders\hbox to 0.7em{\hss.\hss}\hfill\hspace{0.3em}\brfpages{#3}\par}}
% \brftoagroup{title}{first}: a group heading ("Cases:"); the first group carries the "Page(s)" column head.
\newcommand\brftoagroup[2]{%
  \par\needspace{60pt}\ifnum#2=1 \else\vspace{19.6pt}\fi%
  {\singlespacing\noindent\textbf{#1}\ifnum#2=1 \hfill\textbf{\underline{Page(s)}}\fi\par}}
