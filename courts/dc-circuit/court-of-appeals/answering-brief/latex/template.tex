% U.S. Court of Appeals for the District of Columbia Circuit: appellee's (answering) brief, LaTeX engine.
%
% The layout rules are those of typst/cadc-answering-brief.typ (see ../README.md). The body is pandoc's LaTeX output;
% cadc.lua (in this folder) builds the cover, the certificate of parties, the tables, the signature block and the
% certificates from the frontmatter and numbers the headings. The tables of contents and authorities are two-pass:
% each entry writes its page to the .aux or .toc file, and the next run prints the pages (fb render reruns pdflatex
% until they settle). Content-agnostic: no case-specific text, no coordinates, no forced line breaks.
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, list labels, contents rows) is the shared package lib/latex/xengine.sty,
% which fb render inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are
% lib/latex/xengine.lua, run by cadc.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks are
% spaced with \addvspace (the larger of the space below one and above the next).
\documentclass[14pt,letterpaper]{extarticle}
\usepackage[T1]{fontenc}
% FRAP 32(a)(5)(A), Circuit Rule 28: 14-point serif. TeX Gyre Schola (the typst template's face): tgschola ships with
% every TeX Live (package tex-gyre), so the generated .tex compiles with stock pdflatex and no repo-local file.
\usepackage{tgschola}
% FRAP 32(a)(4): letter, 1 in margins; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=36bp]{geometry}
\usepackage{enumitem}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
% Links in the brief are citations (toa:) or none; the fallbacks keep pandoc's output valid.
\providecommand{\href}[2]{#2}
\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}

\makeatletter
% ---------------------------------------------------------------------------
% Type and spacing: line boxes [typst pitch-double, pitch-single, pitch-heading]
% ---------------------------------------------------------------------------
% Double spacing, 26.4 bp boxes, for the body (FRAP 32(a)(4)); single spacing, 17.5 bp boxes, for headings, quotations,
% footnotes, the tables, the cover and the certificates.
\renewcommand\normalsize{\xe@lines{14}{26.4}}
\newcommand\briefsingle{\xe@lines{14}{17.5}}
\renewcommand\footnotesize{\briefsingle}
\renewcommand\small{\briefsingle}
\normalsize
\setlength{\parindent}{0.5in} % after xengine.sty's \raggedright, which sets it to zero
\setcounter{secnumdepth}{0}
\xe@lists{0.5in}
% The page number in the 14 bp type, single-spaced.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\briefsingle\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}
\pagestyle{plain}

% An ellipsis (…) as one character, as typst and Word set it: the T1 fonts have no ellipsis glyph, so three periods in
% the glyph's 1 em, marked in the PDF as the character U+2026 (ActualText, a stock pdfTeX primitive).
\newcommand\briefellipsis{\pdfliteral direct{/Span<</ActualText<FEFF2026>>>BDC}\makebox[1em][s]{.\hfil.\hfil.}\pdfliteral direct{EMC}}
% A glyph whose advance in TeX Live's tgschola (Type 1) differs from the vendored TeX Gyre Schola OpenType that typst
% and LibreOffice set: centered in a box of the OpenType advance, #2..#5 per 1000 em in regular, italic, bold and bold
% italic (cadc.lua writes them for ( ) [ ] / * +).
\def\cadc@it{it}
\def\cadc@pick#1#2{\ifx\f@shape\cadc@it #2\else #1\fi}
\def\cadc@units#1#2#3#4{\if b\expandafter\@car\f@series\@nil\cadc@pick{#3}{#4}\else\cadc@pick{#1}{#2}\fi}
% The bold faces' interword space is the OpenType one, 278 per 1000 em (tgschola's is 287).
\AddToHook{selectfont}[cadc]{\if b\expandafter\@car\f@series\@nil\fontdimen2\font=\dimexpr\fontdimen6\font*278/1000\relax\fi}
\newcommand\cadcg[5]{\leavevmode\hbox to \dimexpr\fontdimen6\font*\cadc@units{#2}{#3}{#4}{#5}/1000\relax{\hss#1\hss}}
% The word that holds such a glyph is marked with its text (ActualText, UTF-16BE hex), so the PDF text is the word.
\newcommand\cadcspan[1]{\leavevmode\pdfliteral direct{/Span<</ActualText<FEFF#1>>>BDC}}
\newcommand\cadcend{\pdfliteral direct{EMC}}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{#1}

% Footnotes [footnote.entry]: 14 point, single spaced, 6.9 bp and the rule above the 7 bp before the first note (the
% 13.9 pt LibreOffice keeps between the text and a page's notes), a 30 % rule, 7 bp above each note;
% the mark (58 % of 14 pt) and a quarter em before the text. LaTeX resets the right skip inside a footnote: it is set
% back here.
\setlength{\skip\footins}{6.9bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}
\xe@footnotemarks{14}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=7bp\relax
  \vskip 0pt\noindent\xe@mark{\@thefnmark}\hskip 0.25em #1}

% Quotations [quote]: single spaced, half an inch in on both sides, 14 bp above and below, 12.6 bp between paragraphs.
\renewenvironment{quote}{%
  \par\addvspace{14bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\briefsingle\parindent=0pt\relax
  \parskip=12.6bp\relax\vskip-\parskip % the first paragraph's skip is the 14 bp above, not more
}{%
  \par\endgroup\addvspace{14bp}%
}

% ---------------------------------------------------------------------------
% Headings. The labels come from cadc.lua; the layout is here. [style-heading]: 21 / 15.4 bp around a section heading,
% 16.8 / 14 bp around a point heading, 15.4 / 12.6 bp around a sub-heading, 16.8 / 12.6 bp around an unnumbered one;
% a heading never ends a page (\nobreak, and no orphans after it).
% ---------------------------------------------------------------------------
\newlength{\briefheadtab}\setlength{\briefheadtab}{0.5in}
% The end of a heading: cadc.lua writes the contents entry right after the heading, so the entry is taken in here and
% the \nobreak follows it (after the entry, a whatsit, the page could otherwise break below the heading).
% A list right after a heading must not open with a break either (typst: the heading is sticky): LaTeX's \@nobreak
% flag makes its first item skip the list's break penalty; the next paragraph clears it.
\newcommand\brief@endhead[1]{\@ifnextchar\addcontentsline{\brief@endtoc{#1}}{\brief@keep{#1}}}
\def\brief@endtoc#1\addcontentsline#2#3#4{\addcontentsline{#2}{#3}{#4}\brief@keep{#1}}
\newcommand\brief@keep[1]{\nobreak\addvspace{#1}\global\@nobreaktrue\everypar{\global\@nobreakfalse\everypar{}}}
% Section headings: bold capitals, centered, 16 bp in 21 bp line boxes.
\newcommand\briefhead[1]{%
  \par\addvspace{21bp}%
  \begingroup\centering\xe@lines{16}{21}\bfseries\MakeUppercase{#1}\par\endgroup
  \brief@endhead{15.4bp}}
% Point headings of the Argument: bold, centered, the label in front.
\newcommand\briefpoint[2]{%
  \par\addvspace{16.8bp}%
  \begingroup\centering\briefsingle\bfseries #1\ #2\par\endgroup
  \brief@endhead{14bp}}
% Sub-headings: the label at indent #1 in a half-inch box, the text hanging under it.
% #2 label, #3 style (bold; bold italic from level 4), #4 text.
\newcommand\briefsub[4]{%
  \par\addvspace{15.4bp}%
  \begingroup\briefsingle\noindent\hangindent=\dimexpr#1+\briefheadtab\relax\hangafter=1\relax
  \hspace*{#1}\makebox[\briefheadtab][l]{#3{#2}}#3{#4}\par\endgroup
  \brief@endhead{12.6bp}}
% Unnumbered sub-headings (addendum): bold, at the margin.
\newcommand\briefplain[1]{%
  \par\addvspace{16.8bp}%
  \begingroup\briefsingle\noindent\textbf{#1}\par\endgroup
  \brief@endhead{12.6bp}}

% ---------------------------------------------------------------------------
% Tables of contents and authorities [listing-row]: the text wraps before a page column (36 bp in the contents, 56 bp
% or the page list's width in the authorities), a dot leader runs to the page.
% ---------------------------------------------------------------------------
% A contents row: #1 the label's indent, #2 label (empty: none), #3 title, #4 page, #5 space above.
\newcommand\brief@colw{36bp} % the page column; the addendum's rows ("Add-12") take the authorities' 56 bp
\newcommand\brief@tocrow[5]{%
  \begingroup\briefsingle\setlength{\xe@col}{\brief@colw}%
  \def\brief@lab{#2}%
  \xe@row{#5}{\dimexpr#1+\briefheadtab\relax}{%
    \hspace*{#1}\ifx\brief@lab\empty\else\makebox[\briefheadtab][l]{#2}\fi #3}{#4}%
  \endgroup}
% A section heading's row (\addcontentsline{toc}{brieftoc}{TITLE}) and a numbered heading's row
% (\addcontentsline{toc}{briefn}{{<label indent>}{<label>}{<title>}}); the addendum's own table (file adt).
\newcommand\l@brieftoc[2]{\brief@tocrow{0in}{}{#1}{#2}{8.4bp}}
\newcommand\l@briefn[2]{\brief@nrow#1{#2}}
\def\brief@nrow#1#2#3#4{\brief@tocrow{#1}{#2}{#3}{#4}{4.9bp}}
\newcommand\l@briefadd[2]{{\def\brief@colw{56bp}\brief@tocrow{0in}{}{#1}{#2}{8.4bp}}}
\newcommand\l@briefaddn[2]{{\def\brief@colw{56bp}\brief@nrow#1{#2}}}
% The column head over the page numbers.
\newcommand\briefTOCpage{\par\addvspace{4.2bp}\begingroup\raggedleft\briefsingle\bfseries\underline{Page}\par\endgroup}
\newcommand\briefTOC[1]{\par\begingroup\briefsingle\@starttoc{#1}\endgroup}

% Table of authorities (Circuit Rule 28(a)(2): every page is listed; "passim" may not be used). A citation writes its
% page to the .aux file (\toamark); the next run reads the marks back (\toaref) and keeps the distinct pages in
% document order.
\newcommand\toamark[1]{\protected@write\@auxout{}{\string\toaref{#1}{\thepage}}}
\newcommand\toaref[2]{%
  \@ifundefined{toacnt@#1}{%
    \expandafter\newcount\csname toacnt@#1\endcsname
    \expandafter\gdef\csname toapg@#1\endcsname{}}{}%
  \@ifundefined{toaseen@#1@#2}{%
    \expandafter\gdef\csname toaseen@#1@#2\endcsname{}%
    \global\advance\csname toacnt@#1\endcsname 1
    \ifnum\csname toacnt@#1\endcsname>1 \expandafter\g@addto@macro\csname toapg@#1\endcsname{, }\fi
    \expandafter\g@addto@macro\csname toapg@#1\endcsname{#2}}{}}
\newcommand\toapages[1]{\@ifundefined{toacnt@#1}{}{\csname toapg@#1\endcsname}}
% One authority [listing-row]: key, the lines before the citation (name and line break, or nothing), citation.
\newlength\brief@w
\newcommand\toaprint[3]{%
  \@ifundefined{toacnt@#1}{}{%
    \begingroup\briefsingle
    \settowidth{\brief@w}{\toapages{#1}}%
    \setlength{\xe@col}{56bp}\ifdim\brief@w>\xe@col\setlength{\xe@col}{\brief@w}\fi
    \xe@row{9.8bp}{0.35in}{#2#3}{\toapages{#1}}%
    \endgroup}}
\newcommand\toaitem[3]{\toaprint{#1}{#2,\newline}{#3}}     % key, name, citation
\newcommand\toaitemnoname[2]{\toaprint{#1}{}{#2}}          % key, citation
% A group title; the first one carries the column head "Page(s)" at the right margin. It stays with its first entry.
\newcommand\toagroup[1]{%
  \par\addvspace{19.6bp}\begingroup\briefsingle\noindent\textbf{#1}\par\endgroup\nobreak\addvspace{2.8bp}}
\newcommand\toagroupfirst[1]{%
  \par\addvspace{14bp}\begingroup\briefsingle\noindent\textbf{#1}\hskip 0pt plus 1fill\textbf{\underline{Page(s)}}\par\endgroup
  \nobreak\addvspace{2.8bp}}

% ---------------------------------------------------------------------------

\newcommand\briefgap[1]{\par\vskip #1bp plus 1fill minus #1bp\relax}
\newcommand\briefrulesmall{\par\hbox to\linewidth{\hss\vrule width 2in height 0.7bp\hss}}
\newcommand\briefdouble{\par\hrule height 1.6bp\vskip 2.5bp\hrule height 0.5bp}

\newenvironment{briefcert}{\par\begingroup\briefsingle\setlength{\parindent}{0pt}}{\par\endgroup}
