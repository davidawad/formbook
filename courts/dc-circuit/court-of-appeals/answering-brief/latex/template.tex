\newcommand\dc@hone[1]{%
  \par\addvspace{1.5em}%
  \begingroup\centering\fontsize{16.1pt}{18pt}\selectfont\bfseries\leavevmode\MakeUppercase{#1}\par\endgroup
  \nopagebreak\addvspace{0.3em}\nobreak}

  \par\addvspace{1.2em}%

  \nopagebreak\addvspace{1em}\nobreak}
\newcommand\dc@hsub[4]{%
  \par\addvspace{1.1em}%
  \begingroup\dc@single\bfseries\ifnum#1>3 \itshape\fi
  \noindent\hangindent=\dimexpr#2+0.5in\relax\hangafter=1\relax
  \hskip#2\relax\hbox to 0.5in{#3\hss}#4\par
  \endgroup

  \begingroup\par\addvspace{0.35em}%
  \noindent\hangindent=\dimexpr#1+0.5in\relax\hangafter=1\relax

% ---------------------------------------------------------------------------
% Table of authorities (Circuit Rule 28(a)(2)): every page of each citation; "passim" or similar may not be used.
% Pages are written to the aux file where each citation starts, then read back in the next pass.
% ---------------------------------------------------------------------------
\ExplSyntaxOn
\cs_new_protected:Npn \dc@toaentry #1 #2
  {
    \seq_if_exist:cF { g_dc_toa_#1_seq } { \seq_new:c { g_dc_toa_#1_seq } }
    \seq_if_in:cnF { g_dc_toa_#1_seq } { #2 }
      { \seq_gput_right:cn { g_dc_toa_#1_seq } { #2 } }
  }
\cs_new:Npn \dc@toapages #1
  {
    \seq_if_exist:cTF { g_dc_toa_#1_seq }
      {
        { \seq_use:cn { g_dc_toa_#1_seq } { , \c_space_tl } }
      }
      { }
  }
\ExplSyntaxOff
