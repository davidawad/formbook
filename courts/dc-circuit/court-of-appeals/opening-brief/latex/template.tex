% U.S. Court of Appeals for the District of Columbia Circuit: appellant's principal brief, LaTeX engine.
%
% Built by tools/fb (fb render): pandoc reads the markdown with the filter cadc.lua (this directory), which writes the
% whole brief (cover, certificate, tables, body, signature, certificates) into the body; this template holds the layout,
% the layout rules of typst/cadc-opening-brief.typ (see ../README.md). pdflatex reruns until the tables' page
% numbers settle. Content-agnostic: no case-specific text, no coordinates, no forced line breaks.
%
% Cross-engine layout (docs/EVAL-LOOP.md, "Cross-engine gate"): this template sets every line and page break where
% the typst template (the reference) sets them. The model itself (no kerning or ligatures, first-fit breaking, fixed
% line boxes, no widows, footnote marks, contents rows) is the shared package lib/latex/xengine.sty, which fb render
% inlines into the generated .tex in place of \usepackage{xengine}; the inline rules are lib/latex/xengine.lua, run by
% cadc.lua. Every size and space here is in bp (14 bp = typst's 14 pt), and blocks are spaced with \addvspace (the
% larger of the space below one and above the next).
\documentclass[14pt,letterpaper]{extarticle}
\usepackage[T1]{fontenc}
% FRAP 32(a)(5)(A), Handbook IX.A.6: 14-point Century. TeX Gyre Schola (the typst template's face): tgschola ships with
% every TeX Live (package tex-gyre), so the generated .tex compiles with stock pdflatex and no repo-local file.
\usepackage{tgschola}
% FRAP 32(a)(4): letter, 1 inch on all four sides; the page number in the bottom margin, as typst sets it.
\usepackage[letterpaper,margin=1in,headheight=0pt,headsep=0pt,footskip=38.4bp]{geometry}
\usepackage{enumitem}
\usepackage{xengine} % lib/latex/xengine.sty, inlined by fb render
% Links in the brief are citations (toa:) or none; the fallbacks keep pandoc's output valid.
\providecommand{\href}[2]{#2}
\providecommand{\hypertarget}[2]{#2}
\providecommand{\hyperlink}[2]{#2}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{#1}

\makeatletter
% ---------------------------------------------------------------------------
% Type and spacing (FRAP 32(a)(4)-(5)): line boxes [typst pitch-body, pitch-single]
% ---------------------------------------------------------------------------
% Double spacing, 25 bp boxes, for the body; single spacing, 21 bp boxes, for quotations, footnotes, the tables, the
% signature and the certificates' signature blocks; the cover in 16.4 bp boxes.
\renewcommand\normalsize{\xe@lines{14}{25}}
\newcommand\dcsingle{\xe@lines{14}{21}}

\normalsize

% The page number in the 14 bp type, single-spaced, centered.
\def\ps@plain{\let\@mkboth\@gobbletwo\let\@oddhead\@empty\let\@evenhead\@empty
  \def\@oddfoot{\reset@font\dcsingle\hfil\thepage\hfil}\let\@evenfoot\@oddfoot}
\pagestyle{plain}

% One face: typst and LibreOffice set TeX Gyre Schola 2.609 (fonts/tex-gyre-schola), whose parentheses, brackets,
% braces, *, +, /, <, =, >, \, | and ellipsis are wider than the Type 1 tgschola of TeX Live (the Adobe New Century
% Schoolbook widths). cadc.lua writes each of them as \dcg{<regular advance>}{<bold advance>}{<glyph>} (thousandths of
% an em; the italics have the upright advances): the glyph scaled across to the OpenType advance less 0.03 em on each
% side (a gap of its own would split "6103(i)(2)" into two words in the PDF's text).
\newcommand\dc@ifbold[2]{\if b\expandafter\@car\f@series\@nil #1\else #2\fi}
\DeclareRobustCommand\dcg[3]{\leavevmode\begingroup
  \setbox0\hbox{#3}\dimen0=\dc@ifbold{#2}{#1}\dimexpr 1em/1000\relax
  \hbox to\dimen0{\hskip0.03em\pdfsave\pdfsetmatrix{\fpeval{(\the\dimen0-\the\dimexpr0.06em\relax)/\the\wd0} 0 0 1}%
    \rlap{\box0}\pdfrestore\hss}\endgroup}
% The bold interword space of the OpenType face is 0.278 em (the Type 1 bold's 0.287 em).
\AddToHook{selectfont}[dcbold]{\dc@ifbold{\fontdimen2\font=0.278\fontdimen6\font}{}}
% An ellipsis (…) as one character, as typst and Word set it: three periods in the glyph's advance (0.86 em, bold
% 1 em), marked in the PDF as U+2026 (ActualText).
\DeclareRobustCommand\dcellipsis{\pdfliteral direct{/Span<</ActualText<FEFF2026>>>BDC}%
  \makebox[\dc@ifbold{1000}{860}\dimexpr 1em/1000\relax][s]{.\hfil.\hfil.}\pdfliteral direct{EMC}}

% Footnotes [footnote.entry]: 14 point, single spaced, 12.6 bp clear of the text, a 30 % rule, 3 bp above each note;
% the mark (58 % of 14 pt) and a quarter em before the text. LaTeX resets the right skip inside a footnote: it is set
% back here.
\setlength{\skip\footins}{12.6bp}
\renewcommand\footnoterule{\vskip 0pt plus 1fill\kern-0.25bp\hrule width 0.3\textwidth height 0.5bp\kern-0.25bp}
\xe@footnotemarks{14}
\renewcommand\@makefntext[1]{\rightskip=0pt plus 1fil\relax\parfillskip=0pt plus 1fil\relax\parskip=3bp\relax
  \vskip 0pt\noindent\xe@mark{\@thefnmark}\hskip 0.25em #1}

% Block quotations: single spaced, half an inch in on both sides, 10 bp above and below (FRAP 32(a)(4)).
\renewenvironment{quote}{%
  \par\addvspace{10bp}%
  \begingroup\leftskip=0.5in\rightskip=\dimexpr0.5in\relax plus 1fil\relax\dcsingle\parindent=0pt\relax
}{%
  \par\endgroup\addvspace{10bp}%
}

% ---------------------------------------------------------------------------
% Headings [typst show heading]: level 1 centered bold 19.5 bp in 29 bp boxes, 24 / 20 bp around; level 2 bold
% 17 bp in 25 bp boxes, 17 / 14 bp around; level 3 bold, half an inch in, single, 12 / 10 bp around. A heading never
% ends a page. Levels 1 and 2 go to the table of contents (typst: outline depth 2); the entry is written inside the
% heading's paragraph.
% ---------------------------------------------------------------------------
\newcommand\dc@keep[1]{\nobreak\addvspace{#1}\global\@nobreaktrue\everypar{\global\@nobreakfalse\everypar{}}}

  \begingroup\centering\xe@lines{19.5}{29}\bfseries #1\addcontentsline{toc}{dcone}{#1}\par\endgroup
  \dc@keep{20bp}}
% A level-1 heading that is not listed (the table of contents itself).
\newcommand\dchoneplain[1]{%

\newcommand\dchtwo[1]{%
  \par\addvspace{17bp}%

\newcommand\dchthree[1]{%
  \par\addvspace{12bp}%
  \begingroup\noindent\leftskip=0.5in\relax\dcsingle\bfseries #1\par\endgroup
  \dc@keep{10bp}}

% Table of authorities (Circuit Rule 28(a)(2): every page, no "passim"). A citation writes its page to the .aux file
% (\toamark); the next run reads the marks back (\toaref) and keeps the distinct pages in document order.
\newcommand\toamark[1]{\protected@write\@auxout{}{\string\toaref{#1}{\thepage}}}
\newcommand\toaref[2]{%
  \@ifundefined{toapg@#1}{\expandafter\gdef\csname toapg@#1\endcsname{}}{}%
  \@ifundefined{toaseen@#1@#2}{%
    \expandafter\gdef\csname toaseen@#1@#2\endcsname{}%
    \expandafter\ifx\csname toapg@#1\endcsname\@empty\else
      \expandafter\g@addto@macro\csname toapg@#1\endcsname{, }\fi
    \expandafter\g@addto@macro\csname toapg@#1\endcsname{#2}}{}}
\newcommand\toapages[1]{\@ifundefined{toapg@#1}{}{\csname toapg@#1\endcsname}}
% One authority: key, its text.
\newlength\dc@w
\newcommand\toaitem[2]{%
  \begingroup\dcsingle
  \settowidth{\dc@w}{\toapages{#1}}%
  \setlength{\xe@col}{36bp}\ifdim\dc@w>\xe@col\setlength{\xe@col}{\dc@w}\fi
  \xe@row{3bp}{0.35in}{#2}{\toapages{#1}}%
  \endgroup}
% A group title (#1: the space above it); it stays with its first entry.
\newcommand\toagroup[2]{%
  \par\addvspace{#1}\begingroup\dcsingle\noindent\textbf{#2}\par\endgroup\nobreak\addvspace{2bp}}
