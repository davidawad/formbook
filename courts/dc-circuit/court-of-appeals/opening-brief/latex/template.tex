\newcommand\dc@hone[1]{%
  \par\addvspace{1.4\baselineskip}%
  \begingroup\centering\dc@single\bfseries\leavevmode#1\par\endgroup
  \nopagebreak\addvspace{1.2\baselineskip}\nobreak}
\newcommand\dc@htwo[1]{%
  \par\addvspace{1.2\baselineskip}%
  \begingroup\raggedright\dc@single\bfseries\leavevmode#1\par\endgroup
  \nopagebreak\addvspace{1em}\nobreak}
\newcommand\dc@hthree[1]{%
  \par\addvspace{1em}%
  \begingroup\noindent\dc@single\bfseries\hskip\dc@half\leavevmode#1\par\endgroup
  \nopagebreak\addvspace{0.9em}\nobreak}
